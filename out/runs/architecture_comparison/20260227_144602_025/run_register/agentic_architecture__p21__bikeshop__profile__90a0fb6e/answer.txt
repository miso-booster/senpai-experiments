test_name: "bike_shop_appointment"

        type: string

  is_starter: true
  fallback: "I'm sorry, I didn't quite catch that. Can you please repeat?"
  output:
    - appointment_confirmation:
        type: string
        description: "Confirmation message for the scheduled appointment."
    - appointment_date:
        type: string
        description: "The date of the scheduled appointment."
    - appointment_time:
        type: string
        description: "The time of the scheduled appointment."

  number: combinations(0.5, pairwise)  # Using pairwise combinations for testing

      - change language:
          - Spanish
          - French
