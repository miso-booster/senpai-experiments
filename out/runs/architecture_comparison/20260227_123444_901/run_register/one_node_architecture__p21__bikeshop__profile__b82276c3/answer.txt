```yaml
test_name: "wheel_change_appointment"

llm:
  temperature: 0.7
  model: gpt-4o
  model_prov: openai
  format:
    type: text

user:
  language: English
  role: "A customer looking to set an appointment for changing bike wheels."
  context:
    - "The user prefers morning appointments."
    - "The user is flexible with dates but prefers weekends."
  goals:
    - "Set an appointment for changing wheels on {{appointment_date}} at {{appointment_time}}."
    - "Ask if there are any available slots on {{alternative_date}} at {{alternative_time}}."
    - "Inquire about the estimated time for the wheel change service."
    - appointment_date:
        function: random()
        type: string
        data:
          - "Saturday"
          - "Sunday"
          - "Next Monday"
          - "Next Tuesday"
    - appointment_time:
        function: random()
        type: string
        data:
          - "9:00 AM"
          - "10:30 AM"
          - "1:00 PM"
          - "3:00 PM"
    - alternative_date:
        function: random()
        type: string
        data:
          - "Saturday"
          - "Sunday"
          - "Next Monday"
          - "Next Tuesday"
    - alternative_time:
        function: random()
        type: string
        data:
          - "9:00 AM"
          - "10:30 AM"
          - "1:00 PM"
          - "3:00 PM"

chatbot:
  is_starter: false
  fallback: "I'm sorry, I didn't understand that. Can you please rephrase?"
  output:
    - appointment_confirmation:
        type: string
        description: "Confirmation of the appointment date and time."
    - estimated_time:
        type: string
        description: "Estimated time for the wheel change service."

conversation:
  number: 2
  goal_style:
    steps: 4
  interaction_style:
    - random:
      - long phrases
      - single questions
```
