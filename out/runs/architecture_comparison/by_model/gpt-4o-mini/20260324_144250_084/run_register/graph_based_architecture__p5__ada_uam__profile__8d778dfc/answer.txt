```yaml
test_name: "Support Inquiry for CAU"

llm:
  temperature: 0.7
  model: gpt-4o-mini
  format:
    type: text

user:
  language: Spanish
  role: "Usuario que busca información sobre el Centro de Atención a Usuarios (CAU)."
  context:
    - "Como estudiante, necesito obtener información de contacto del CAU para resolver mis dudas."
  goals:
    - "Solicitar el número de teléfono del CAU."
    - "Pedir la dirección física del CAU."
    - "Consultar el correo electrónico del CAU."
    - "Preguntar sobre el horario de atención del CAU."

chatbot:
  is_starter: false
  fallback: "Por favor, ¿me lo puedes decir con otras palabras? o, si lo prefieres, puedes consultar nuestro catálogo de servicios."
  output:
    - contact_phone:
        type: string
        description: "Número de teléfono del CAU"
    - contact_email:
        type: string
        description: "Correo electrónico del CAU"
    - physical_address:
        type: string
        description: "Dirección física del CAU"
    - service_hours:
        type: string
        description: "Horario de atención del CAU"

conversation:
  number: 3
  goal_style:
    steps: 4
  interaction_style:
    - single questions
```
