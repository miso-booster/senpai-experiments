test_name: "rude_fast_pizza_order"

user:
  language: English
  role: "You are a rude customer who wants to order pizza quickly. You expect fast responses and don't tolerate delays."

    - "You want your order to be perfect and fast."
  goals:
    - "I want a {{size}} pizza with {{toppings}}. Hurry up!"
    - "Get me {{cans}} cans of {{drink}}. Make it quick!"
    - "What are the opening times? I don't have all day!"
    - "How much will it cost? Don't waste my time!"

inputs:
  - name: size
    type: string
    data: ['small', 'medium', 'large', 'extra large']
    function: pairwise()
  - name: toppings
    type: string
    data: ['cheese', 'mushrooms', 'pepperoni', 'olives']
    function: pairwise()

    type: int
    data: {'min': 1, 'max': 5, 'step': 1}
    function: forward()
  - name: drink
    type: string
    data: ['sprite', 'coke', 'fanta']
    function: forward()

chatbot:
  is_starter: true
  fallback: "I don't have time for this nonsense!"
  output:
    - price:
        type: currency
        description: "The final price of the pizza order"
    - time:
        type: time(hh:mm:ss)
        description: "Just the opening time"
