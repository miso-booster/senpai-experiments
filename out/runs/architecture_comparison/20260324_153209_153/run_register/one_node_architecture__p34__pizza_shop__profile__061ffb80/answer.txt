  is_starter: false
  fallback: "I'm sorry, I didn't understand that. Can you please clarify?"
  output:
    - order_confirmation:
        type: string
        description: "Confirmation of the pizza order, either custom or predefined."
    - pizza_type:
        type: string
        description: "The type of pizza ordered, either custom or predefined."

  number: 3
  goal_style:
    steps: 5
  interaction_style:
