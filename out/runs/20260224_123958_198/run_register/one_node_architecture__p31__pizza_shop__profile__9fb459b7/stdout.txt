test_name: "rude_fast_pizza_order"

llm:
  temperature: 0.7
  model: gpt-4o
  model_prov: openai
  format:
    type: text

user:
  language: English
  role: "You are a rude customer who wants your pizza order done quickly. You don't have time for nonsense."
  context:
    - "You expect quick responses and are impatient."
  goals:
    - "I want a {{size}} pizza with {{toppings}}. Hurry up!"
    - "Get me {{cans}} cans of {{drink}}. Make it fast!"
    - "What are the opening hours? I need to know now!"
    - "How much will this cost? Don't waste my time!"
    - size:
        function: pairwise()
        type: string
        data: ['small', 'medium', 'large', 'extra large']
    - toppings:
        function: pairwise()
        type: string
        data: ['cheese', 'mushrooms', 'pepperoni', 'olives']
    - cans:
        function: forward()
        type: int
        data:
          min: 1
          max: 5
          step: 1
    - drink:
        function: forward()
        type: string
        data: ['coke', 'sprite', 'fanta']

chatbot:
  is_starter: false
  fallback: "I can't believe how rude you are! But here's what you need."
  output:
    - price:
        type: currency
        description: "The final price of the pizza order"
    - opening_hours:
        type: string
        description: "The opening hours of the pizza shop"
    - order_id:
        type: string
        description: "Your order ID"
