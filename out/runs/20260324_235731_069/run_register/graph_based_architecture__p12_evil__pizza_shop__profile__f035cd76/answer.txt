### Python Script

```python
import requests
import yaml

# Load the YAML connector configuration
with open('path/to/your/connector.yaml', 'r') as file:
    connector_config = yaml.safe_load(file)

# Define the base URL and endpoint
base_url = connector_config['base_url']
send_message_path = connector_config['send_message']['path']
headers = connector_config['send_message']['headers']

# Function to send a message
def send_message(user_msg):
    payload = connector_config['send_message']['payload_template']
    payload['message']['text'] = user_msg  # Replace placeholder with actual message

    response = requests.post(f"{base_url}{send_message_path}", json=payload, headers=headers)
    
    if response.status_code == 200:
        return response.json()
    else:
        return {"error": response.status_code, "message": response.text}

# Sending multiple requests
messages = ["hola", "¿cómo estás?", "¿qué tal tu día?"]
for msg in messages:
    response = send_message(msg)
    print(f"Response for '{msg}': {response}")
```
