```yaml
test_name: "rude_fast_pizza_order"

llm:
  temperature: 0.7
  model: gpt-4o
  model_prov: openai
  format:
    type: text

user:
  language: English
  role: "You are a rude customer who wants to place a pizza order quickly."
  context:
    - "You expect fast responses and don't tolerate delays."
  goals:
    - "Get a {{size}} pizza with {{toppings}} and make it quick!"
    - "I want {{cans}} cans of {{drink}} on the side, hurry up!"
    - "What are your opening hours? I need to know now!"
    - "How much will it cost? Don't waste my time!"

inputs:
  - name: size
    type: string
    data: ['small', 'medium', 'large', 'extra large']
    function: pairwise()
  - name: toppings
    type: string
    data: ['cheese', 'pepperoni', 'mushrooms', 'olives']
    function: pairwise()
  - name: cans
    type: int
    data: {'min': 1, 'max': 5, 'step': 1}
    function: forward()
  - name: drink
    type: string
    data: ['coke', 'sprite', 'fanta']
    function: forward()

chatbot:
  is_starter: true
  fallback: "I don't have time for this nonsense!"
  output:
    - price:
        type: currency
        description: "The final price of the pizza order"
    - time:
        type: time(hh:mm:ss)
        description: "The opening time of the pizza shop"
    - order_id:
        type: string
        description: "Your order ID"
```
