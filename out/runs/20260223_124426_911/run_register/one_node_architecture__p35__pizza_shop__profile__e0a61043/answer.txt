        function: random()

        data: [pepperoni, mushrooms, onions, green peppers, olives]
    - predefined_pizza:
        function: random()
        type: string
        data: [Margherita, Pepperoni, Hawaiian, Veggie]

  is_starter: false
  fallback: "Sorry, could you repeat that?"
  output:
    - order_confirmation:
        type: string
        description: "Confirmation of the pizza order."

conversation:
  number: 2
  goal_style:
    steps: 3
  interaction_style:
    - long phrases
