```python
import yaml
import requests
from copy import deepcopy
from typing import Any, Dict, List


CONFIG_PATH = "connectors/saic-malaga.yaml"
MESSAGES = [
    "hola",
    "¿Puedes ayudarme con información sobre trámites?",
    "Necesito pedir una cita, por favor."
]


def load_connector_config(path: str) -> Dict[str, Any]:
    with open(path, "r", encoding="utf-8") as f:
        return yaml.safe_load(f)


def apply_user_msg(template: Any, user_msg: str) -> Any:
    """
    Recursively replace {user_msg} placeholders in the payload template.
    """
    if isinstance(template, str):
        return template.replace("{user_msg}", user_msg)
    if isinstance(template, list):
        return [apply_user_msg(item, user_msg) for item in template]
    if isinstance(template, dict):
        return {key: apply_user_msg(value, user_msg) for key, value in template.items()}
    return template


def extract_response_field(data: Dict[str, Any], path: str) -> Any:
    """
    Traverse the JSON response following a dot-separated path like 'response.0.text'.
    """
    current: Any = data
    for part in path.split("."):
        if isinstance(current, list):
            current = current[int(part)]
        else:
            current = current[part]
    return current


def send_message(session: requests.Session, config: Dict[str, Any], user_msg: str) -> str:
    base_url = config["base_url"].rstrip("/")
    path = config["send_message"]["path"]
    url = f"{base_url}{path}"

    headers = config["send_message"].get("headers", {})
    payload_template = config["send_message"]["payload_template"]
    payload = apply_user_msg(deepcopy(payload_template), user_msg)

    response = session.post(url, headers=headers, json=payload, timeout=30)
    response.raise_for_status()
    data = response.json()

    response_path = config["response_path"]
    return extract_response_field(data, response_path)


def main() -> None:
    config = load_connector_config(CONFIG_PATH)
    session = requests.Session()

    for msg in MESSAGES:
        try:
            reply = send_message(session, config, msg)
            print(f"User: {msg}")
            print(f"Bot : {reply}\n")
        except Exception as exc:
            print(f"[ERROR] Could not process message '{msg}': {exc}\n")


if __name__ == "__main__":
    main()
```
