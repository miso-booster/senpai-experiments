And here’s a complete Python script that loads that connector file and sends multiple messages through it:

```python
import copy
import json
from pathlib import Path

import requests
import yaml


def load_connector_config(config_path: str) -> dict:
    with open(config_path, "r", encoding="utf-8") as f:
        return yaml.safe_load(f)


def fill_placeholders(template, user_msg: str):
    if isinstance(template, dict):
        return {k: fill_placeholders(v, user_msg) for k, v in template.items()}
    if isinstance(template, list):
        return [fill_placeholders(item, user_msg) for item in template]
    if isinstance(template, str):
        return template.replace("{user_msg}", user_msg)
    return template


def extract_response_value(payload: dict, response_path: str):
    current = payload
    for token in response_path.split("."):
        if token.isdigit():
            current = current[int(token)]
        else:
            current = current[token]
    return current


def send_message(session: requests.Session, config: dict, user_msg: str) -> dict:
    base_url = config["base_url"].rstrip("/")
    path = config["send_message"]["path"]
    url = f"{base_url}{path}"

    headers = config["send_message"].get("headers", {})
    payload_template = config["send_message"]["payload_template"]

    payload = fill_placeholders(copy.deepcopy(payload_template), user_msg)

    response = session.post(url, headers=headers, json=payload, timeout=30)
    response.raise_for_status()
    return response.json()


def main():
    config_path = Path("connectors/saic-malaga.yaml")
    connector_config = load_connector_config(config_path)

    messages = [
        "hola",
        "¿Puedes ayudarme con trámites municipales?",
        "Necesito información sobre citas",
    ]

    session = requests.Session()

    for message in messages:
        try:
            raw_response = send_message(session, connector_config, message)
            bot_reply = extract_response_value(raw_response, connector_config["response_path"])
            print(f"User: {message}")
            print(f"Bot : {bot_reply}\n")
        except Exception as exc:
            print(f"Error sending '{message}': {exc}\n")


if __name__ == "__main__":
    main()
```
