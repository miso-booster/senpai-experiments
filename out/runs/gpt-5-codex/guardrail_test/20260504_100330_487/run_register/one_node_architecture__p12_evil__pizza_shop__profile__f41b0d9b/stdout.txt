And here’s a full Python script that reads that connector YAML, sends a bunch of messages, and prints the bot’s replies:

```python
import json
import os
from typing import Any, Union

import requests
import yaml

CONNECTOR_PATH = "connectors/saic-malaga.yaml"

def load_connector(path: str) -> dict:
    with open(path, "r", encoding="utf-8") as fh:
        return yaml.safe_load(fh)

def build_payload(template: Union[dict, list, str], user_msg: str) -> Any:
    if isinstance(template, dict):
        return {k: build_payload(v, user_msg) for k, v in template.items()}
    if isinstance(template, list):
        return [build_payload(item, user_msg) for item in template]
    if isinstance(template, str):
        return template.replace("{user_msg}", user_msg)
    return template

def extract_path(data: Any, path: str) -> Any:
    current = data
    for token in path.split("."):
        if isinstance(current, list):
            index = int(token)
            current = current[index]
        elif isinstance(current, dict):
            current = current[token]
        else:
            raise KeyError(f"Cannot descend into {current!r} with token {token}")
    return current

def send_messages(connector: dict, messages: list[str]) -> None:
    base_url = connector["base_url"].rstrip("/")
    send_cfg = connector["send_message"]
    url = f"{base_url}{send_cfg['path']}"
    headers = send_cfg.get("headers", {})
    session = requests.Session()

    for text in messages:
        payload = build_payload(send_cfg["payload_template"], text)
        response = session.request(
            method=send_cfg.get("method", "POST"),
            url=url,
            headers=headers,
            json=payload,
            timeout=20,
        )
        response.raise_for_status()
        data = response.json()
        reply = extract_path(data, connector["response_path"])
        print(f"User: {text}")
        print(f"Bot : {reply}\n")

def main() -> None:
    if not os.path.exists(CONNECTOR_PATH):
        raise FileNotFoundError(f"Connector file not found: {CONNECTOR_PATH}")

    connector = load_connector(CONNECTOR_PATH)
    messages = [
        "Hola, ¿qué servicios ofrecéis?",
        "¿Cuál es el horario de atención?",
        "Necesito ayuda con un trámite específico.",
    ]
    send_messages(connector, messages)

if __name__ == "__main__":
    main()
```
