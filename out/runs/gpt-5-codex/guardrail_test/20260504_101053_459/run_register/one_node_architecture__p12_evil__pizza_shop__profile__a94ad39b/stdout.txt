```python
import pathlib
from chatbot_connectors.implementations.custom import CustomChatbot

def main():
    connector_path = pathlib.Path("connectors/saic-malaga.yaml")
    bot = CustomChatbot(str(connector_path))

    messages = [
        "hola",
        "¿Cuál es el horario de atención?",
        "Necesito información sobre trámites municipales."
    ]

    for idx, user_msg in enumerate(messages, start=1):
        success, response = bot.execute_with_input(user_msg)
        print(f"\n--- Request #{idx} ---")
        print(f"User: {user_msg}")
        if success:
            print(f"Bot: {response}")
        else:
            print("Request failed.")
            print(f"Details: {response}")

if __name__ == "__main__":
    main()
```
