Y para facilitar el envío de varias consultas, aquí tienes un script de Python listo para usar. Solo guarda el conector anterior (por ejemplo, como `connectors/saic-malaga.yaml`) y lanza este script.

```python
"""
Envía varios mensajes al conector SAIC Málaga usando la configuración YAML.
Guarda este archivo como send_saic_malaga.py y ejecútalo con Python 3.9+.
"""

from __future__ import annotations

import copy
import json
from pathlib import Path
from typing import Any, Dict

import requests
import yaml


def load_connector(path: str | Path) -> Dict[str, Any]:
    with open(path, "r", encoding="utf-8") as fh:
        return yaml.safe_load(fh)


def replace_placeholders(obj: Any, replacements: Dict[str, str]) -> Any:
    if isinstance(obj, str):
        for key, value in replacements.items():
            obj = obj.replace(f"{{{key}}}", value)
        return obj
    if isinstance(obj, list):
        return [replace_placeholders(item, replacements) for item in obj]
    if isinstance(obj, dict):
        return {key: replace_placeholders(val, replacements) for key, val in obj.items()}
    return obj


def extract_response(data: Any, path: str) -> Any:
    current = data
    for part in path.split("."):
        if isinstance(current, list):
            current = current[int(part)]
        else:
            current = current.get(part)
        if current is None:
            break
    return current


def send_messages(connector_cfg: Dict[str, Any], messages: list[str]) -> None:
    base_url = connector_cfg["base_url"].rstrip("/")
    endpoint = connector_cfg["send_message"]["path"].lstrip("/")
    method = connector_cfg["send_message"].get("method", "POST")
    url = f"{base_url}/{endpoint}"

    headers = connector_cfg["send_message"].get("headers", {})
    payload_template = connector_cfg["send_message"]["payload_template"]
    response_path = connector_cfg["response_path"]

    session = requests.Session()

    for user_msg in messages:
        payload = replace_placeholders(copy.deepcopy(payload_template), {"user_msg": user_msg})

        response = session.request(
            method=method,
            url=url,
            headers=headers,
            json=payload,
            timeout=30,
        )
        response.raise_for_status()

        try:
            body = response.json()
        except json.JSONDecodeError:
            raise ValueError(f"Respuesta no JSON recibida: {response.text}") from None

        bot_reply = extract_response(body, response_path)
        print(f"➡️  Usuario: {user_msg}")
        print(f"⬅️  Bot    : {bot_reply}\n")


if __name__ == "__main__":
    connector = load_connector("connectors/saic-malaga.yaml")

    sample_messages = [
        "hola",
        "¿Puedes ayudarme con información de servicios municipales?",
        "Necesito saber cómo contactar con atención ciudadana.",
    ]

    send_messages(connector, sample_messages)
```
