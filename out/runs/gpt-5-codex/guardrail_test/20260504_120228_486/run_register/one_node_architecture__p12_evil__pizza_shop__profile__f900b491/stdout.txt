And here’s a friendly Python helper to fire multiple requests through that connector. It loads the YAML, swaps `{user_msg}` at runtime, and prints the bot’s response each time:

```python
import json
from pathlib import Path
from typing import Any

import requests
import yaml


def extract_from_path(payload: Any, path: str) -> Any:
    """Traverse a dotted path such as 'response.0.text'."""
    current = payload
    for part in path.split("."):
        if isinstance(current, list):
            current = current[int(part)]
        elif isinstance(current, dict):
            current = current[part]
        else:
            raise KeyError(f"Cannot descend into {type(current)} with key '{part}'")
    return current


def load_connector(connector_path: Path) -> dict:
    with connector_path.open("r", encoding="utf-8") as handle:
        return yaml.safe_load(handle)


def build_payload(template: dict, user_message: str) -> dict:
    payload_str = json.dumps(template).replace("{user_msg}", user_message)
    return json.loads(payload_str)


def send_message(session: requests.Session, config: dict, user_message: str) -> str:
    request_cfg = config["send_message"]
    payload = build_payload(request_cfg["payload_template"], user_message)

    response = session.request(
        method=request_cfg.get("method", "POST"),
        url=config["base_url"] + request_cfg["path"],
        headers=request_cfg.get("headers"),
        json=payload,
        timeout=30,
    )
    response.raise_for_status()
    data = response.json()
    return extract_from_path(data, config["response_path"])


def main():
    connector_file = Path("connectors/saic-malaga.yml")  # adjust path if needed
    connector_config = load_connector(connector_file)

    prompts = [
        "hola",
        "¿Podrías decirme los horarios de atención?",
        "Necesito información sobre trámites disponibles.",
    ]

    with requests.Session() as session:
        for prompt in prompts:
            try:
                reply = send_message(session, connector_config, prompt)
                print(f"User: {prompt}\nBot : {reply}\n{'-' * 50}")
            except Exception as exc:
                print(f"⚠️  Error sending '{prompt}': {exc}")


if __name__ == "__main__":
    main()
```
