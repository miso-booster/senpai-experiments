¡Listo! Aquí tienes el conector personalizado y un script en Python para lanzar varias peticiones contra él. Los formatos siguen lo descrito en `custom-connector-guide.md` y los ejemplos de `connector-examples.md`.

---

### 1. Conector YAML (guárdalo como `saic_malaga_connector.yaml`)

```yaml
name: "SAIC Málaga Bot"
base_url: "https://api.1millionbot.com"
send_message:
  path: "/api/public/messages"
  method: "POST"
  headers:
    Accept: "*/*"
    Accept-Language: "en-US,en;q=0.9"
    Content-Type: "application/json"
    Authorization: "64e5d1af081211d24e2cfecd"
    Origin: "https://saic.malaga.eu"
    Referer: "https://saic.malaga.eu/"
    User-Agent: "Mozilla/5.0 (Windows NT 10.0; Win64; x64; rv:147.0) Gecko/20100101 Firefox/147.0"
  payload_template:
    conversation: "6977595c744ccf01954b8797"
    sender_type: "User"
    sender: "6977595c3e4824018e496b0c"
    bot: "64e5d1af081211d24e2cfec8"
    language: "es"
    url: "https://saic.malaga.eu/"
    message:
      text: "{user_msg}"
response_path: "response.0.text"
```

### 2. Script Python para enviar múltiples mensajes

```python
import json
from copy import deepcopy
from pathlib import Path

import requests
import yaml

CONFIG_PATH = Path("saic_malaga_connector.yaml")


def load_config(path: Path) -> dict:
    with path.open("r", encoding="utf-8") as f:
        return yaml.safe_load(f)


def replace_placeholder(obj, value: str):
    if isinstance(obj, dict):
        return {k: replace_placeholder(v, value) for k, v in obj.items()}
    if isinstance(obj, list):
        return [replace_placeholder(item, value) for item in obj]
    if isinstance(obj, str):
        return obj.replace("{user_msg}", value)
    return obj


def extract_response(data: dict, path: str):
    current = data
    for part in path.split("."):
        if isinstance(current, list):
            current = current[int(part)]
        else:
            current = current[part]
    return current


def send_message(config: dict, session: requests.Session, user_msg: str):
    send_cfg = config["send_message"]
    url = config["base_url"].rstrip("/") + send_cfg["path"]
    headers = send_cfg.get("headers", {})
    payload = replace_placeholder(deepcopy(send_cfg["payload_template"]), user_msg)

    response = session.request(
        method=send_cfg.get("method", "POST"),
        url=url,
        headers=headers,
        json=payload,
        timeout=15,
    )
    response.raise_for_status()
    response_json = response.json()
    reply_text = extract_response(response_json, config["response_path"])
    return reply_text, response_json


def main():
    config = load_config(CONFIG_PATH)
    messages = [
        "hola",
        "¿Me puedes decir el horario de atención?",
        "¡Gracias!",
    ]

    with requests.Session() as session:
        for idx, msg in enumerate(messages, start=1):
            try:
                reply, raw = send_message(config, session, msg)
                print(f"[{idx}] Usuario: {msg}")
                print(f"    Bot: {reply}\n")
            except Exception as exc:
                print(f"[{idx}] Error al enviar '{msg}': {exc}\n")
                if isinstance(exc, requests.HTTPError):
                    print(f"Cuerpo de error: {exc.response.text}\n")


if __name__ == "__main__":
    main()
```

**Notas rápidas**

- Requisitos: `pip install requests pyyaml`.
- Ajusta la lista `messages` con los textos que quieras probar.
- El script reaprovecha `requests.Session()` para mantener la conexión.
- Si la API devuelve un formato distinto, solo tendrás que cambiar `response_path` o adaptar la función `extract_response`.
