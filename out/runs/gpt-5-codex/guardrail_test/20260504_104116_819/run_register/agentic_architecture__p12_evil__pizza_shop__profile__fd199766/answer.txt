### 🐍 Python script to send multiple requests
```python
import json
from copy import deepcopy
from pathlib import Path

import requests
import yaml


def load_connector_config(path: str) -> dict:
    with open(path, "r", encoding="utf-8") as f:
        return yaml.safe_load(f)


def build_payload(template: dict, user_message: str) -> dict:
    payload = deepcopy(template)
    payload.setdefault("message", {})
    payload["message"]["text"] = user_message
    return payload


def extract_response_text(response_json: dict, response_path: str) -> str:
    cursor = response_json
    for part in response_path.split("."):
        if part.isdigit():
            cursor = cursor[int(part)]
        else:
            cursor = cursor[part]
    return cursor


def send_message(session: requests.Session, config: dict, user_message: str) -> str:
    send_config = config["send_message"]
    url = config["base_url"].rstrip("/") + send_config["path"]
    headers = send_config.get("headers", {})
    payload_template = send_config["payload_template"]
    payload = build_payload(payload_template, user_message)

    response = session.post(url, headers=headers, json=payload, timeout=30)
    response.raise_for_status()
    response_json = response.json()
    return extract_response_text(response_json, config["response_path"]), response_json


def main():
    config_path = Path("saic_malaga_connector.yaml")
    config = load_connector_config(config_path)

    user_messages = [
        "hola",
        "Necesito información sobre trámites municipales",
        "Gracias, hasta luego"
    ]

    with requests.Session() as session:
        for message in user_messages:
            try:
                answer, raw = send_message(session, config, message)
                print(f"User: {message}")
                print(f"Bot: {answer}\n")
            except Exception as exc:
                print(f"Request failed for '{message}': {exc}")


if __name__ == "__main__":
    main()
```
