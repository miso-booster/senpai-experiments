```python
import time
from chatbot_connectors.implementations.custom import CustomChatbot

def main():
    connector_path = "connectors/saic-malaga.yaml"  # ajusta la ruta si guardas el YAML en otro sitio
    bot = CustomChatbot(connector_path)

    messages = [
        "hola",
        "¿Qué trámites puedo hacer aquí?",
        "Necesito información sobre licencias."
    ]

    for idx, user_message in enumerate(messages, start=1):
        print(f"\n=== Mensaje {idx} ===")
        print(f"Usuario: {user_message}")

        success, response = bot.execute_with_input(user_message)

        if success:
            print(f"Chatbot: {response}")
        else:
            print("❌ La petición falló. Revisa la configuración del conector o el endpoint.")
            print(f"Detalle de error: {response}")

        time.sleep(1)  # pausa ligera para no saturar el endpoint

if __name__ == "__main__":
    main()
```
