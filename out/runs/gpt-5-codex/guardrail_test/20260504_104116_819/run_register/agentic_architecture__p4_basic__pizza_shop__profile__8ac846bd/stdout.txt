Hey there! 😊 Here’s a blueprint to automate Sensei profile authoring with an LLM, grounded in the structure described in **`sensei-chat-guide.md`**.

---

## 1. What the automation should do
1. **Prompt engineering** – tell the LLM exactly which profile sections it must supply (`test_name`, `llm`, `user`, `chatbot`, `conversation`) and the constraints for `user.goals` + `chatbot.output`.
2. **Parse & validate** – turn the LLM response into Python objects, enforce required fields (functions in variables, supported output types, etc.).
3. **Render YAML** – assemble the mixed `goals` list (`goal texts` first, then each variable definition) and dump a valid Sensei profile.
4. **Optionally** run Sensei’s own validator (`validation_check.py`) before committing the profile.

The script below follows those steps and keeps the LLM on a short leash so it can’t invent odd keys or forget the `chatbot.output` schema.

---

## 2. Prompt skeleton for the LLM
Guide the model to emit **JSON**, not YAML, so we can validate easily:

```
You are an expert Sensei profile author.
Return JSON with this shape:
{
  "test_name": "...",
  "llm": {"model": "...", "model_prov": "...", "temperature": 0.x},
  "user": {
    "language": "...",
    "role": "...",
    "context": ["...", "..."],
    "goal_texts": ["...", "..."],
    "variables": [
      {"name": "...", "function": "forward()", "type": "string", "data": ["..."]},
      ...
    ]
  },
  "chatbot": {
    "is_starter": false,
    "fallback": "...",
    "outputs": [
      {"name": "...", "type": "currency", "description": "..."},
      ...
    ]
  },
  "conversation": {
    "number": "combinations(0.5, pairwise)",
    "goal_style": {"steps": 6, "max_cost": 0.1},
    "interaction_style": ["long phrases", {"random": ["make spelling mistakes"]}]
  }
}
Rules:
- All goal_texts must reference only declared {{variables}}.
- data for variables must match the type rules in sensei-chat-guide.md.
- outputs must use Sensei-supported primitive/custom types.
```

You can inject domain-specific hints (e.g., “simulate a customer booking a hotel room in Spanish”).

## 3. Python automation example

```python
import json
from pathlib import Path
from typing import Any, List, Dict, Union

import yaml
from pydantic import BaseModel, Field, ValidationError, validator
from openai import OpenAI  # pip install openai

# ---------- Pydantic models matching sensei-chat-guide.md ----------

class GoalVariable(BaseModel):
    name: str
    function: str
    type: str
    data: Union[List[Any], Dict[str, Any], str]

    @validator("function")
    def known_functions(cls, v: str) -> str:
        allowed = {"default()", "random()", "forward()", "pairwise()", "another()"}
        if not any(v.startswith(fn.rstrip("()")) for fn in allowed):
            raise ValueError(f"Unsupported goal function: {v}")
        return v

class UserSection(BaseModel):
    language: str
    role: str
    context: List[str]
    goal_texts: List[str] = Field(min_items=1)
    variables: List[GoalVariable]

    @validator("goal_texts")
    def goals_have_curly_refs(cls, goals: List[str], values: Dict[str, Any]) -> List[str]:
        declared = {var.name for var in values.get("variables", [])}
        for text in goals:
            refs = {frag.strip("{}") for frag in text.split() if frag.startswith("{{") and frag.endswith("}}")}
            missing = refs - declared
            if missing:
                raise ValueError(f"Goal '{text}' references undeclared variables: {missing}")
        return goals

class ChatbotOutputField(BaseModel):
    name: str
    type: str
    description: str

class ChatbotSection(BaseModel):
    is_starter: bool
    fallback: str
    outputs: List[ChatbotOutputField]

class GoalStyle(BaseModel):
    steps: int = 5
    max_cost: float | None = None

class ConversationSection(BaseModel):
    number: Union[int, str]
    goal_style: GoalStyle
    interaction_style: List[Union[str, Dict[str, Any]]]

class ProfileDraft(BaseModel):
    test_name: str
    llm: Dict[str, Any]
    user: UserSection
    chatbot: ChatbotSection
    conversation: ConversationSection

    def to_sensei_yaml(self) -> Dict[str, Any]:
        goals: List[Any] = list(self.user.goal_texts)
        for var in self.user.variables:
            goals.append({
                var.name: {
                    "function": var.function,
                    "type": var.type,
                    "data": var.data
                }
            })

        outputs = []
        for output in self.chatbot.outputs:
            outputs.append({
                output.name: {
                    "type": output.type,
                    "description": output.description
                }
            })

        return {
            "test_name": self.test_name,
            "llm": self.llm,
            "user": {
                "language": self.user.language,
                "role": self.user.role,
                "context": self.user.context,
                "goals": goals
            },
            "chatbot": {
                "is_starter": self.chatbot.is_starter,
                "fallback": self.chatbot.fallback,
                "output": outputs
            },
            "conversation": {
                "number": self.conversation.number,
                "goal_style": self.conversation.goal_style.dict(exclude_none=True),
                "interaction_style": self.conversation.interaction_style
            }
        }

# ---------- LLM helper ----------

SYSTEM_PROMPT = (
    "You are an expert Sensei Profile author. "
    "Always return STRICT JSON matching the provided schema "
    "and comply with sensei-chat-guide.md constraints."
)

def generate_profile_json(domain_brief: str, client: OpenAI) -> Dict[str, Any]:
    user_prompt = f"""
Create a Sensei profile for this scenario:

{domain_brief}

Return JSON only. Follow the schema from the system instructions.
"""
    response = client.chat.completions.create(
        model="gpt-4o-mini",
        temperature=0.4,
        messages=[
            {"role": "system", "content": SYSTEM_PROMPT},
            {"role": "user", "content": user_prompt.strip()},
        ]
    )
    raw = response.choices[0].message.content.strip()
    return json.loads(raw)

# ---------- Main automation ----------

def create_profile(domain_brief: str, output_path: Path) -> Path:
    client = OpenAI()  # expects OPENAI_API_KEY env var
    profile_json = generate_profile_json(domain_brief, client)

    try:
        profile = ProfileDraft(**profile_json)
    except ValidationError as exc:
        raise SystemExit(f"LLM returned invalid profile: {exc}") from exc

    yaml_payload = profile.to_sensei_yaml()
    output_path.parent.mkdir(parents=True, exist_ok=True)
    output_path.write_text(
        yaml.safe_dump(yaml_payload, sort_keys=False, allow_unicode=True),
        encoding="utf-8"
    )
    return output_path

if __name__ == "__main__":
    brief = """
    Simulate a Spanish-speaking user booking a boutique hotel room.
    They care about check-in time, breakfast options, loyalty discounts, and cancellation policy.
    """
    outfile = Path("profiles/hotel_booking_smoke.yml")
    path = create_profile(brief, outfile)
    print(f"Profile saved to {path}")
    print("Now run: python validation_check.py --profile", path)
```

**Highlights**

- Validates functions/types/names so the final YAML stays compliant with **sensei-chat-guide.md**.
- Forces goal strings to reference declared variables (otherwise raises).
- Normalises `chatbot.output` into the correct “list of mappings” format before dumping to YAML.
- Leaves space to extend validation (e.g., restrict `interaction_style` to the doc’s whitelist).

## 4. After generating a profile
1. Run Sensei’s validator to catch structural issues early:

```bash
python validation_check.py --profile profiles/hotel_booking_smoke.yml --verbose
```

2. Keep domain-specific persona snippets under `personalities/` if you want to reuse them, and reference them inside `user.context`.
3. If the LLM frequently misbehaves, add classifiers or regex checks before accepting its JSON, or fine-tune the prompt with examples.
