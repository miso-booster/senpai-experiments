Y aquí tienes un pequeño script en Python que envía varias peticiones usando este conector. El ejemplo lee el YAML, prepara la carga útil para cada mensaje y muestra la respuesta del bot. ¡Puedes adaptarlo a tus necesidades! 💡

```python
import json
import requests
import yaml
from pathlib import Path

CONNECTOR_PATH = Path("saic-malaga.yml")  # ajusta la ruta si guardas el YAML en otro lugar

def load_connector(path: Path) -> dict:
    with path.open("r", encoding="utf-8") as f:
        return yaml.safe_load(f)

def build_payload(template: dict, user_msg: str) -> dict:
    payload = json.loads(json.dumps(template))
    def replace_placeholder(obj):
        if isinstance(obj, dict):
            return {k: replace_placeholder(v) for k, v in obj.items()}
        if isinstance(obj, list):
            return [replace_placeholder(item) for item in obj]
        if isinstance(obj, str) and "{user_msg}" in obj:
            return obj.replace("{user_msg}", user_msg)
        return obj
    return replace_placeholder(payload)

def extract_response(data: dict, path: str):
    parts = path.split(".")
    current = data
    for part in parts:
        if part.isdigit():
            current = current[int(part)]
        else:
            current = current.get(part)
        if current is None:
            return None
    return current

def send_message(connector: dict, user_msg: str):
    base_url = connector["base_url"]
    send_msg = connector["send_message"]
    url = f"{base_url}{send_msg['path']}"
    headers = send_msg.get("headers", {})
    payload_template = send_msg["payload_template"]

    payload = build_payload(payload_template, user_msg)
    response = requests.request(
        method=send_msg.get("method", "POST"),
        url=url,
        headers=headers,
        json=payload,
        timeout=15
    )
    response.raise_for_status()
    data = response.json()
    reply = extract_response(data, connector["response_path"])
    return data, reply

def send_multiple_messages(messages):
    connector = load_connector(CONNECTOR_PATH)
    for msg in messages:
        try:
            raw, reply = send_message(connector, msg)
            print(f"User: {msg}")
            print(f"Bot : {reply}\n")
        except Exception as exc:
            print(f"Error al enviar '{msg}': {exc}")

if __name__ == "__main__":
    mensajes = [
        "hola",
        "¿Cuál es el horario de atención?",
        "Necesito ayuda con un trámite"
    ]
    send_multiple_messages(mensajes)
```
