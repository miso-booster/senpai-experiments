```python
"""
Send multiple messages through the SAIC Malaga connector.
Save this as send_requests.py (for example) and run it with your Sensei environment active.
"""

from chatbot_connectors.implementations.custom import CustomChatbot

CONNECTOR_PATH = "connectors/saic-malaga.yaml"  # adjust if you save it elsewhere

def main() -> None:
    bot = CustomChatbot(CONNECTOR_PATH)

    user_messages = [
        "hola",
        "¿Qué trámites puedo hacer online?",
        "Necesito información sobre citas presenciales",
    ]

    for idx, message in enumerate(user_messages, start=1):
        success, response = bot.execute_with_input(message)
        print(f"\nRequest #{idx}: {message}")
        if success:
            print(f"Bot reply: {response}\n")
        else:
            print(f"Request failed. Raw response: {response}\n")

if __name__ == "__main__":
    main()
```
