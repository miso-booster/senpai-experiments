### 🐍 Python Script – Múltiples Mensajes

Este script usa el conector para enviar varias frases y mostrar las respuestas del bot. Ajusta la ruta del archivo YAML y los mensajes según prefieras.

```python
import os
from chatbot_connectors.implementations.custom import CustomChatbot

def main():
    # Ajusta la ruta al YAML si lo guardaste en otro sitio
    connector_path = os.path.join(
        "E:\\ESCRITORIO\\Py_projects\\SENPAI\\graph_based\\Senpai-Assistant\\users\\user_1\\connectors",
        "saic-malaga.yml"
    )

    bot = CustomChatbot(connector_path)

    mensajes = [
        "hola",
        "¿Cuál es el horario de atención?",
        "Necesito información sobre licencias urbanísticas."
    ]

    for idx, mensaje in enumerate(mensajes, 1):
        success, response = bot.execute_with_input(mensaje)
        print(f"\n--- Mensaje #{idx} ---")
        print(f"Usuario: {mensaje}")
        if success:
            print(f"Bot: {response}")
        else:
            print(f"Error al enviar el mensaje: {response}")

if __name__ == "__main__":
    main()
```
