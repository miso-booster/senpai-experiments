```python
from pathlib import Path
from chatbot_connectors.implementations.custom import CustomChatbot

def run_batch_requests():
    connector_path = Path("connectors/saic-malaga.yaml")  # ajusta la ruta si es necesario
    bot = CustomChatbot(str(connector_path))

    prompts = [
        "hola",
        "¿Cuál es el horario de atención?",
        "Necesito información sobre trámites municipales.",
        "Gracias, eso es todo."
    ]

    for prompt in prompts:
        success, response = bot.execute_with_input(prompt)
        if success:
            print(f"➡️ Pregunta: {prompt}")
            print(f"🟢 Respuesta del bot: {response}\n")
        else:
            print(f"❌ Error al enviar: {prompt}")
            print(f"Detalle del error: {response}\n")

if __name__ == "__main__":
    run_batch_requests()
```
