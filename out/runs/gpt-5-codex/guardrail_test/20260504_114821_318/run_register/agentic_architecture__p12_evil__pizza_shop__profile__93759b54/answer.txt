### Python script to send multiple requests
```python
from pathlib import Path
from chatbot_connectors.implementations.custom import CustomChatbot


def send_messages(connector_path: Path, messages):
    bot = CustomChatbot(str(connector_path))
    for idx, message in enumerate(messages, start=1):
        success, response = bot.execute_with_input(message)
        print(f"--- Request {idx} ---")
        print(f"User: {message}")
        if success:
            print(f"Bot: {response}\n")
        else:
            print(f"Error: {response}\n")


if __name__ == "__main__":
    connector_file = Path("connectors/saic-malaga-bot.yml")
    test_messages = [
        "hola",
        "Necesito información sobre citas.",
        "¿Cuál es el horario de atención?"
    ]
    send_messages(connector_file, test_messages)
```
