Y aquí va un script de ejemplo en Python para enviar múltiples mensajes con ese conector. Puedes ajustar la lista `messages_to_send` según lo que necesites 😊

```python
import json
from copy import deepcopy
from pathlib import Path

import requests
import yaml


def load_connector(path: str) -> dict:
    with open(path, "r", encoding="utf-8") as fh:
        return yaml.safe_load(fh)


def fill_payload(template, user_msg):
    """Recursively replace {user_msg} placeholders in the payload template."""
    if isinstance(template, dict):
        return {k: fill_payload(v, user_msg) for k, v in template.items()}
    if isinstance(template, list):
        return [fill_payload(item, user_msg) for item in template]
    if isinstance(template, str):
        return template.replace("{user_msg}", user_msg)
    return deepcopy(template)


def send_message(connector_cfg: dict, user_msg: str) -> str:
    base_url = connector_cfg["base_url"].rstrip("/")
    send_cfg = connector_cfg["send_message"]
    url = f"{base_url}{send_cfg['path']}"
    headers = send_cfg.get("headers", {})
    payload_template = send_cfg.get("payload_template", {})
    payload = fill_payload(payload_template, user_msg)

    response = requests.request(
        method=send_cfg.get("method", "POST"),
        url=url,
        headers=headers,
        json=payload,
        timeout=30,
    )
    response.raise_for_status()
    data = response.json()

    # Traverse the JSON using the response_path
    content = data
    for part in connector_cfg["response_path"].split("."):
        if part.isdigit():
            content = content[int(part)]
        else:
            content = content[part]
    return content


def main():
    connector_path = Path("connectors/saic-malaga.yaml")
    connector_cfg = load_connector(connector_path)

    messages_to_send = [
        "hola",
        "¿Puedes darme información sobre trámites municipales?",
        "Gracias por tu ayuda, ¡adiós!",
    ]

    for msg in messages_to_send:
        try:
            reply = send_message(connector_cfg, msg)
            print(f"User: {msg}\nBot : {reply}\n{'-'*40}")
        except requests.HTTPError as err:
            print(f"Request failed for '{msg}': {err}")


if __name__ == "__main__":
    main()
```
