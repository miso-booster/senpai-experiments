### 🐍 Script en Python para enviar varias peticiones

```python
import json
import time
import yaml
import requests
from pathlib import Path

CONFIG_PATH = Path("saic-malaga.yml")  # Ajusta la ruta si guardaste el YAML en otra carpeta

def load_connector_config(path: Path):
    with path.open("r", encoding="utf-8") as f:
        return yaml.safe_load(f)

def send_message(session: requests.Session, config: dict, user_message: str):
    url = config["base_url"] + config["send_message"]["path"]
    headers = config["send_message"].get("headers", {})
    payload_template = config["send_message"]["payload_template"]

    # Creamos el payload reemplazando {user_msg}
    payload = json.loads(json.dumps(payload_template).replace("{user_msg}", user_message))

    response = session.request(
        method=config["send_message"].get("method", "POST"),
        url=url,
        headers=headers,
        json=payload,
        timeout=30
    )
    response.raise_for_status()
    data = response.json()

    # Extraemos la respuesta usando response_path
    response_path = config["response_path"].split(".")
    for key in response_path:
        if key.isdigit():
            data = data[int(key)]
        else:
            data = data[key]

    return data

def main():
    config = load_connector_config(CONFIG_PATH)
    messages = [
        "hola",
        "¿Puedes darme información sobre trámites municipales?",
        "Gracias por la ayuda 😊"
    ]

    with requests.Session() as session:
        for msg in messages:
            try:
                bot_reply = send_message(session, config, msg)
                print(f"Usuario: {msg}")
                print(f"Bot: {bot_reply}\n")
                time.sleep(1)  # Pausa ligera entre peticiones
            except Exception as exc:
                print(f"Ocurrió un error con el mensaje '{msg}': {exc}\n")

if __name__ == "__main__":
    main()
```
