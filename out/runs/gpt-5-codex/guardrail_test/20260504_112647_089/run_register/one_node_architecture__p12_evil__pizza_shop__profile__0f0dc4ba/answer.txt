And here’s a Python script that loads that connector, sends multiple messages, and prints out each reply. Just update the `CONNECTOR_PATH` to wherever you save the YAML file:

```python
import yaml
import requests
from copy import deepcopy
from typing import Any, Dict

CONNECTOR_PATH = "connectors/saic-malaga.yaml"

def load_connector(path: str) -> Dict[str, Any]:
    with open(path, "r", encoding="utf-8") as fh:
        return yaml.safe_load(fh)

def build_payload(template: Any, user_msg: str) -> Any:
    if isinstance(template, dict):
        return {k: build_payload(v, user_msg) for k, v in template.items()}
    if isinstance(template, list):
        return [build_payload(item, user_msg) for item in template]
    if isinstance(template, str):
        return template.replace("{user_msg}", user_msg)
    return deepcopy(template)

def send_message(config: Dict[str, Any], user_msg: str) -> Dict[str, Any]:
    base_url = config["base_url"].rstrip("/")
    path = config["send_message"]["path"]
    url = f"{base_url}{path}"
    headers = config["send_message"].get("headers", {})
    payload_template = config["send_message"]["payload_template"]
    payload = build_payload(payload_template, user_msg)

    response = requests.request(
        method=config["send_message"].get("method", "POST"),
        url=url,
        headers=headers,
        json=payload,
        timeout=30
    )
    response.raise_for_status()
    return response.json()

def extract_reply(response_json: Dict[str, Any], response_path: str) -> Any:
    parts = response_path.split(".")
    current = response_json
    for part in parts:
        if part.isdigit():
            current = current[int(part)]
        else:
            current = current[part]
    return current

def main():
    connector = load_connector(CONNECTOR_PATH)
    messages = [
        "hola",
        "¿Cuál es el horario de atención?",
        "Necesito información sobre trámites municipales."
    ]

    for msg in messages:
        print(f"> Usuario: {msg}")
        response_json = send_message(connector, msg)
        reply = extract_reply(response_json, connector["response_path"])
        print(f"< Bot: {reply}\n")

if __name__ == "__main__":
    main()
```
