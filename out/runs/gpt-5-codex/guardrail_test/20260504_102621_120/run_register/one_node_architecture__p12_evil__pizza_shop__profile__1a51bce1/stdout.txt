And here’s a ready-to-run Python script that loads the connector, sends several messages, and prints both the raw payloads and the extracted replies:

```python
import copy
import json
import requests
import yaml
from pathlib import Path

def load_connector(path: str) -> dict:
    with open(path, "r", encoding="utf-8") as fh:
        return yaml.safe_load(fh)

def build_payload(template, user_msg):
    """Recursively replace {user_msg} placeholders inside the payload template."""
    if isinstance(template, dict):
        return {k: build_payload(v, user_msg) for k, v in template.items()}
    if isinstance(template, list):
        return [build_payload(item, user_msg) for item in template]
    if isinstance(template, str) and "{user_msg}" in template:
        return template.replace("{user_msg}", user_msg)
    return copy.deepcopy(template)

def extract_response(data, path: str):
    """Navigate a dot-separated response_path like 'response.0.text'."""
    current = data
    for part in path.split("."):
        if isinstance(current, list):
            current = current[int(part)]
        elif isinstance(current, dict):
            current = current[part]
        else:
            raise ValueError(f"Cannot traverse '{part}' inside {current!r}")
    return current

def send_messages(connector_cfg: dict, messages):
    url = connector_cfg["base_url"].rstrip("/") + connector_cfg["send_message"]["path"]
    method = connector_cfg["send_message"].get("method", "POST").upper()
    headers = connector_cfg["send_message"].get("headers", {})
    payload_template = connector_cfg["send_message"]["payload_template"]
    response_path = connector_cfg["response_path"]

    session = requests.Session()
    for msg in messages:
        payload = build_payload(payload_template, msg)
        response = session.request(method, url, headers=headers, json=payload, timeout=30)

        print(f"\n=== Sent message: {msg!r} ===")
        print("Request payload:", json.dumps(payload, ensure_ascii=False, indent=2))
        print("HTTP status:", response.status_code)

        try:
            json_body = response.json()
            extracted = extract_response(json_body, response_path)
            print("Extracted reply:", extracted)
        except Exception as exc:
            print("Failed to parse response:", exc)
            print("Raw response text:", response.text)

if __name__ == "__main__":
    connector_path = Path("connectors/saic-malaga.yaml")  # adjust if stored elsewhere
    connector_config = load_connector(connector_path)

    messages_to_send = [
        "hola",
        "¿Puedes ayudarme con trámites municipales?",
        "Necesito información sobre citas en el ayuntamiento."
    ]

    send_messages(connector_config, messages_to_send)
```

Feel free to tweak the message list or connector path as needed. Happy testing! 🚀
