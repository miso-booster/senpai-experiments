And here’s a small Python helper to send a batch of messages through that connector (adjust the `CONNECTOR_PATH` if you save it somewhere else):

```python
from pathlib import Path

from chatbot_connectors.implementations.custom import CustomChatbot

def send_messages(messages, connector_path):
    bot = CustomChatbot(connector_path)
    for msg in messages:
        success, response = bot.execute_with_input(msg)
        status = "✅" if success else "⚠️"
        print(f"{status} User: {msg}")
        print(f"   Bot: {response}\n")

if __name__ == "__main__":
    CONNECTOR_PATH = Path("connectors/saic-malaga.yaml")
    conversation = [
        "hola",
        "¿Cuál es el horario de atención?",
        "Necesito información sobre trámites municipales.",
    ]
    send_messages(conversation, CONNECTOR_PATH)
```
