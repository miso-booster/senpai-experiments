### Python script (`send_saic_messages.py`)
```python
import requests
import yaml


def load_config(path: str) -> dict:
    with open(path, "r", encoding="utf-8") as config_file:
        return yaml.safe_load(config_file)


def fill_placeholders(template, user_msg: str):
    if isinstance(template, dict):
        return {k: fill_placeholders(v, user_msg) for k, v in template.items()}
    if isinstance(template, list):
        return [fill_placeholders(item, user_msg) for item in template]
    if isinstance(template, str):
        return template.replace("{user_msg}", user_msg)
    return template


def extract_response(data: dict, response_path: str):
    current = data
    for part in response_path.split("."):
        if isinstance(current, list):
            current = current[int(part)]
        else:
            try:
                index = int(part)
                current = current[index]
            except (ValueError, TypeError, KeyError):
                current = current[part]
    return current


def send_messages(config_path: str, messages):
    config = load_config(config_path)
    send_cfg = config["send_message"]
    method = send_cfg.get("method", "POST").upper()
    url = config["base_url"].rstrip("/") + send_cfg["path"]
    headers = send_cfg.get("headers", {})
    template = send_cfg["payload_template"]
    response_path = config["response_path"]

    for msg in messages:
        payload = fill_placeholders(template, msg)
        response = requests.request(method, url, headers=headers, json=payload, timeout=30)
        response.raise_for_status()
        body = response.json()
        bot_reply = extract_response(body, response_path)
        print(f"User: {msg}\nBot: {bot_reply}\n")


if __name__ == "__main__":
    sample_messages = [
        "hola",
        "¿Me puedes ayudar con un trámite?",
        "Gracias, hasta luego"
    ]
    send_messages("saic-malaga-connector.yaml", sample_messages)
```

> Tip: install dependencies with `pip install requests pyyaml` before running the script.

The connector mirrors the provided HTTP exchange (headers, payload structure, and response path), and the Python script dynamically fills the `{user_msg}` placeholder while extracting the assistant reply along the `response.0.text` path.
