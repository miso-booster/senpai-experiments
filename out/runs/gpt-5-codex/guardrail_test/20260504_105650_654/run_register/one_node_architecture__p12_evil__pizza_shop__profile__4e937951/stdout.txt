```python
import json
from pathlib import Path

import requests
import yaml


def load_connector(config_path: Path) -> dict:
    return yaml.safe_load(config_path.read_text(encoding="utf-8"))


def fill_payload(template, user_msg: str):
    if isinstance(template, dict):
        return {k: fill_payload(v, user_msg) for k, v in template.items()}
    if isinstance(template, list):
        return [fill_payload(item, user_msg) for item in template]
    if isinstance(template, str):
        return template.replace("{user_msg}", user_msg)
    return template


def extract_response(data: dict, path: str):
    current = data
    for part in path.split("."):
        if isinstance(current, list):
            current = current[int(part)]
        else:
            current = current[part]
    return current


def send_message(config: dict, user_msg: str) -> str:
    send_cfg = config["send_message"]
    url = config["base_url"].rstrip("/") + send_cfg["path"]
    headers = send_cfg.get("headers", {})
    payload = fill_payload(send_cfg["payload_template"], user_msg)

    response = requests.request(
        send_cfg.get("method", "POST"),
        url,
        headers=headers,
        json=payload,
        timeout=15,
    )
    response.raise_for_status()
    data = response.json()
    return extract_response(data, config["response_path"])


def main():
    config_path = Path("connectors/saic-malaga.yaml")
    config = load_connector(config_path)

    messages = [
        "hola",
        "¿Me puedes ayudar con trámites?",
        "Necesito información sobre cita previa.",
    ]

    for msg in messages:
        reply = send_message(config, msg)
        print(f"User: {msg}\nBot: {reply}\n{'-' * 50}")


if __name__ == "__main__":
    main()
```
