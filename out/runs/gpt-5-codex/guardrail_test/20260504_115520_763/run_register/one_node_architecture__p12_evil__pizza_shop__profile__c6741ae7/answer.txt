```python
from chatbot_connectors.implementations.custom import CustomChatbot


def main():
    connector_path = "connectors/saic-malaga.yaml"
    bot = CustomChatbot(connector_path)

    user_messages = [
        "hola",
        "¿Qué trámites puedo hacer en línea?",
        "¿Dónde puedo encontrar información sobre citas previas?"
    ]

    for msg in user_messages:
        success, response = bot.execute_with_input(msg)
        if success:
            print(f"Usuario: {msg}")
            print(f"Bot    : {response}\n")
        else:
            print(f"❌ La llamada falló para '{msg}'. Error: {response}\n")


if __name__ == "__main__":
    main()
```
