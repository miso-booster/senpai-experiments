```yaml
# File: profiles/bike_fit_discomfort_information_seeker.yaml
test_name: bike_fit_discomfort_information_seeker

llm:
  model: gpt-4o-mini
  temperature: 0.5
  format:
    type: text

user:
  language: English
  role: >
    A cyclist experiencing discomfort during rides who wants guidance on whether
    the bike fit or worn components might be the cause.
  context:
    - You rely on your bike for regular commuting and weekend rides.
    - You have not had a professional fitting before.
    - You're hoping to resolve the issue soon but are not ready to schedule an appointment yet.
  goals:
    - "Explain that after {{ride_duration}} you feel {{discomfort_area}} and ask what might be causing it."
    - "Ask what kinds of assessments or fitting services the shop offers to diagnose discomfort."
    - "Ask if the shop provides inspections for worn components like the saddle, drivetrain, or handlebars."
    - "Request suggestions for adjustments or quick checks you can try before committing to a service."
    - "Ask about pricing or time estimates for any recommended fitting or inspection services."
    - ride_duration:
        type: string
        function: random()
        data:
          - longer weekend rides
          - 30-40 minute commutes
          - short training sessions
    - discomfort_area:
        type: string
        function: random()
        data:
          - knee pain
          - numb hands
          - lower back soreness
          - saddle discomfort
          - shoulder tension

chatbot:
  is_starter: false
  fallback: "I'm sorry it's a little loud in my shop, can you say that again?"
  output:
    - discomfort_guidance:
        type: string
        description: "Advice the chatbot gives about potential causes of the rider's discomfort."
    - fitting_services_overview:
        type: string
        description: "Details about bike fitting or assessment services the shop offers."
    - component_inspection_info:
        type: string
        description: "Information about inspections or diagnostics for worn components."
    - adjustment_tips:
        type: string
        description: "Specific suggestions for adjustments or self-checks the rider can try."
    - service_pricing_estimate:
        type: currency
        description: "Pricing or time estimates for the recommended fitting or inspection services."

conversation:
  number: 3
  max_cost: 0.5
  goal_style:
    all_answered:
      export: false
      limit: 20
  interaction_style:
    - default
```
