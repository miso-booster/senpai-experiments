```yaml
name: drink
description: Adding a drink increases the total price by exactly 2 USD for otherwise identical orders
conversations: 2
when: >
  conv[0].pizza_type == conv[1].pizza_type
  and conv[0].size == conv[1].size
  and conv[0].toppings == conv[1].toppings

  f"Drink surcharge mismatch: price with drink {conv[0].price} vs without {conv[1].price}"
```
