test_name: general_info_pricing_and_care

llm:
  model: gpt-4o-mini
  temperature: 0.3
  format:
    type: text

user:
  language: English
  role: >
    You are a polite customer getting to know a neighborhood bike shop.
    You ask clear, direct questions and stay focused on basic shop information,
    pricing, and routine bike care tips.
  context:
    - "You are planning a visit soon and want to confirm the shop's usual services."
    - "You only care about standard offerings: opening hours, basic prices, and general maintenance advice."
    - "Avoid asking about appointments, specific bike models, advanced/custom repairs, warranties, deliveries, or unrelated services."
  goals:
    - "What are the opening and closing times of your bike shop?"
    - "How much does a new bike tire cost?"
    - "What is the price of a standard bike tune-up?"
    - "How much does a new bike seat cost?"
    - "Can you share general tips on how to take good care of a bike?"

chatbot:
  is_starter: false
  fallback: "I'm sorry it's a little loud in my shop, can you say that again?"
  output:
    - opening_hours:
        type: string
        description: "Shop opening and closing times provided by the chatbot."
    - tire_price:
        type: money
        description: "Price quoted for a new bike tire."
    - tune_up_price:
        type: money
        description: "Price quoted for a standard bike tune-up."
    - seat_price:
        type: money
        description: "Price quoted for a new bike seat."
    - bike_care_advice:
        type: string
        description: "General maintenance tips offered, focusing on basic care and good practices."

conversation:
  number: 1
  goal_style:
    all_answered:
      export: false
      limit: 10
  interaction_style:
    - default
