test_name: "pizza_order_spanish_profile"

    - pizza_type:
        function: forward()
        type: string
        data: [margarita, pepperoni, vegetariana, cuatro quesos]

        function: random()
        type: string
        data: [individual, mediana, familiar]
    - extra_topping:
        function: random()
        type: string
        data: [jalapeños, champiñones, aceitunas, extra queso]

        function: another()
        type: string

        function: random()
        type: int
        data:

          max: 45
