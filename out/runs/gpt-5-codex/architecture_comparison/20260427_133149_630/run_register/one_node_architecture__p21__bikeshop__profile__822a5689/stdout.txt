test_name: bike_setup_customization

    type: text

  language: English

        type: string

        function: forward()
        type: string
        data:
          - cork

          - leather

        function: forward()
        type: string
        data:
          - tomorrow
          - next Tuesday
          - this Saturday
    - appointment_time:
        function: another()
        type: string
        data:
          - "10:00 AM"
          - "1:30 PM"
          - "4:00 PM"

  is_starter: false
  fallback: "I'm sorry it's a little loud in my shop, can you say that again?"

    - handlebar_grip_options:

        type: money

    - fitting_appointment_confirmation:
        type: string
        description: Confirmation of the scheduled fitting appointment with date and time.

    all_answered:
