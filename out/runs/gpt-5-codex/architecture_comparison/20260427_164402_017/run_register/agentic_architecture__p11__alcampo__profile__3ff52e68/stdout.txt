```yaml
test_name: "subscription_services_information"

llm:
  temperature: 0.6
  model: gpt-4o
  model_prov: openai
  format:
    type: text

user:
  language: English
  role: >
    You are a potential customer evaluating subscription services and want
    to understand which plan offers the best value for your needs.
  context:
    - "You are comparing subscription tiers to decide whether to sign up."
    - "You care about plan benefits, pricing transparency, and cancellation flexibility."
  goals:
    - "Ask for an overview of the available subscription services."
    - "Request a detailed breakdown of the benefits included in each subscription plan."
    - "Clarify the pricing structure and billing cycles for the subscription options."
    - "Understand the cancellation terms and any refund policies for the subscriptions."

chatbot:
  is_starter: false
  fallback: "I'm sorry, could you clarify what you need about our subscriptions?"
  output:
    - plans_offered:
        type: list[string]
        description: "Names or descriptions of the subscription plans mentioned by the chatbot."
    - plan_benefits:
        type: string
        description: "Summary of the benefits associated with each subscription plan."
    - pricing_details:
        type: string
        description: "Pricing and billing cycle information provided for the subscriptions."
    - cancellation_policy:
        type: string
        description: "Details about cancellation terms or refund policies."

conversation:
  number: 3
  goal_style:
    all_answered:
      limit: 12
  interaction_style:
    - single questions
```
