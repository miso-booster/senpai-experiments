```yaml
test_name: "bike_shop_wheel_service_dates"

llm:
  temperature: 0.6
  model: gpt-4o
  model_prov: openai
  format:
    type: text

user:
  language: English
  role: >
    You are an avid cyclist trying to schedule a wheel replacement appointment at a local bike shop.
    You’re specific about dates, times, and turnaround expectations, and you expect a clear confirmation.
  context:
    - "You can bring the bike to the downtown shop in person."
    - "You recently noticed wear on both wheels and want them swapped before an upcoming ride."
    - "You prefer to receive SMS updates once the booking is confirmed."
  goals:
    - "Explain that you need to change both wheels on your {{bike_type}} and would like to book a service appointment."
    - "Ask if the shop has availability on {{appointment_date}} around {{appointment_time}}."
    - "If that slot is unavailable, request {{backup_date}} at around {{backup_time}} instead."
    - "Mention you have a {{availability_constraint}} and need the bike ready accordingly."
    - "Request a confirmation message with any reference number or further instructions."
    - bike_type:
        function: random()
        type: string
        data:
          - road bike
          - mountain bike
          - gravel bike
          - commuter bike
    - appointment_date:
        function: pairwise()
        type: string
        data:
          - next Tuesday
          - April 22

          - May 3rd

        function: pairwise()
        type: string
        data:
          - 9:30 AM
          - 11:45 AM
          - 4:00 PM
          - 18:30
    - backup_date:
        function: another()
        type: string
        data:
          - the following day

    - backup_time:
        function: random()
        type: string

        function: random()

          - can only visit after work

  is_starter: false

  output:

        type: date("any format the chatbot uses")
```
