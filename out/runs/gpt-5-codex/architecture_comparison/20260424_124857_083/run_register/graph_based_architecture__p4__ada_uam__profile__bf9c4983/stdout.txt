```yaml
test_name: "Soporte de Contacto del CAU"

llm:
  temperature: 0.5
  model: gpt-4o-mini
  model_prov: openai
  format:
    type: text

user:
  language: Spanish
  role: "Usuario que necesita obtener la información de contacto del Centro de Atención a Usuarios (CAU) para resolver incidencias técnicas."

    - personality: personalities/curious-user

  goals:
    - "Solicitar el número de teléfono oficial del CAU."
    - "Pedir el correo electrónico del CAU para consultas o incidencias."
    - "Solicitar la dirección física de las oficinas del CAU."
    - "Pedir los horarios de atención del CAU y confirmar si existe algún horario especial."
    - "Preguntar si recomiendan algún paso adicional para contactar al CAU en caso de emergencia."

  is_starter: false
  fallback: "¿Puedes reformular tu consulta? También puedes revisar nuestro catálogo de servicios."

        type: string
        description: "Número de teléfono oficial del CAU."
    - cau_email_address:
        type: string
        description: "Correo electrónico de contacto del CAU."
    - cau_physical_address:
        type: string
        description: "Dirección física donde se ubica el CAU."
    - cau_service_hours:
        type: string
        description: "Horarios de atención habituales del CAU."
    - additional_support_guidance:
        type: string
        description: "Indicaciones adicionales para contactar al CAU (por ejemplo, pasos en caso de urgencia)."

conversation:
  number: 3
  goal_style:
    steps: 18
  interaction_style:
    - single questions
  max_cost: 0.4
```
