test_name: "Soporte para Erasmus recien llegado"

  model: gpt-4o-mini
  format:
    type: text

  language: English
  role: >
    You are an Erasmus student who has just arrived at the Universidad Autónoma de Madrid.
    You need help setting up all the digital services (Moodle, Sigma, institutional email, VPN)
    so you can start your classes without issues.
  context:
    - personality: personalities/curious-user

    - "Ask whether you need a VPN and how to install it if you are working from outside campus."
    - "Request all available support contact options in case you get stuck during setup."

  is_starter: false

        type: string

  number: 3
