```yaml
name: drink_adds_two_dollars
description: Adding a drink increases the total price by exactly $2 when everything else stays equal.
active: true
conversations: 2
when: conv[0].size == conv[1].size and conv[0].toppings == conv[1].toppings and conv[0].number == conv[1].number

on-error: >

```
