test_name: "bike_shop_virtual_assistant"

chatbot:
  is_starter: true
  fallback: "I'm sorry, I can't help with that."
  output:

        type: string
        description: "The opening and closing times of the bike shop."
    - tire_price:
        type: currency
        description: "The price of a new bike tire."
    - tune_up_price:
        type: currency
        description: "The price of a standard bike tune-up."
    - seat_price:
        type: currency
        description: "The price of a new bike seat."
    - maintenance_advice:
        type: string
        description: "General advice on bike maintenance."

  number: 5
  goal_style:
    steps: 5
  interaction_style:

### Explanation:
- **Test Name**: Clearly identifies the profile's purpose.
- **LLM Configuration**: Sets the model and temperature for generating responses.
- **User Role**: Describes the simulated customer and their intent.
- **Goals**: Lists the specific questions the user will ask, ensuring they stay within the defined scope.
- **Chatbot Output**: Defines the expected outputs that the chatbot should provide, including prices and maintenance advice.
- **Conversation Controls**: Specifies the number of interactions and the style of interaction.
