Now, let’s create a complete Python script that will send multiple requests to this connector. This script will use the `requests` library to handle the HTTP requests. Make sure you have it installed in your environment (`pip install requests`).

Here's the Python script:

```python
import requests
import json

# Define the connector configuration
connector_config = {
    "base_url": "https://api.1millionbot.com",
    "path": "/api/public/messages",
    "headers": {
        "Content-Type": "application/json",
        "Authorization": "64e5d1af081211d24e2cfecd",
        "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64; rv:147.0) Gecko/20100101 Firefox/147.0",
        "Accept": "*/*",
        "Accept-Language": "en-US,en;q=0.9",
        "Accept-Encoding": "gzip, deflate, br, zstd",
        "Referer": "https://saic.malaga.eu/",
        "Origin": "https://saic.malaga.eu",
        "Sec-GPC": "1",
        "Connection": "keep-alive",
        "Sec-Fetch-Dest": "empty",
        "Sec-Fetch-Mode": "cors",
        "Sec-Fetch-Site": "cross-site",
        "Priority": "u=4",
        "Pragma": "no-cache",
        "Cache-Control": "no-cache"
    }
}

# Function to send a message
def send_message(user_msg):
    url = f"{connector_config['base_url']}{connector_config['path']}"
    payload = {
        "conversation": "6977595c744ccf01954b8797",
        "sender_type": "User",
        "sender": "6977595c3e4824018e496b0c",
        "bot": "64e5d1af081211d24e2cfec8",
        "language": "es",
        "url": "https://saic.malaga.eu/",
        "message": {
            "text": user_msg
        }
    }
    
    response = requests.post(url, headers=connector_config['headers'], data=json.dumps(payload))
    
    if response.status_code == 200:
        print("Response:", response.json())
    else:
        print("Error:", response.status_code, response.text)

# Sending multiple requests
messages = ["hola", "¿cómo estás?", "¿qué puedes hacer por mí?"]
for msg in messages:
    send_message(msg)
```
