### Python Script to Send Multiple Requests

Here's a complete Python script that uses the connector to send multiple requests:

```python
import requests
import json

# Define the connector configuration
connector_config = {
    "base_url": "https://api.1millionbot.com",
    "send_message": {
        "path": "/api/public/messages",
        "headers": {
            "Content-Type": "application/json",
            "Authorization": "64e5d1af081211d24e2cfecd"
        }
    }
}

# Function to send a message
def send_message(conversation_id, sender_id, bot_id, user_msg):
    url = f"{connector_config['base_url']}{connector_config['send_message']['path']}"
    headers = connector_config['send_message']['headers']
    
    payload = {
        "conversation": conversation_id,
        "sender_type": "User",
        "sender": sender_id,
        "bot": bot_id,
        "language": "es",
        "url": "https://saic.malaga.eu/",
        "message": {
            "text": user_msg
        }
    }
    
    response = requests.post(url, headers=headers, json=payload)
    return response.json()

# Example usage
if __name__ == "__main__":
    conversation_id = "6977595c744ccf01954b8797"
    sender_id = "6977595c3e4824018e496b0c"
    bot_id = "64e5d1af081211d24e2cfec8"
    
    messages = ["hola", "¿cómo estás?", "¿qué puedes hacer?"]
    
    for msg in messages:
        response = send_message(conversation_id, sender_id, bot_id, msg)
        print(f"Response for '{msg}': {response}")
```
