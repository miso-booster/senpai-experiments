```python
import requests
import json

# Define the connector configuration
connector_config = {
    "base_url": "https://api.1millionbot.com",
    "send_message": {
        "path": "/api/public/messages",
        "method": "POST",
        "headers": {
            "Content-Type": "application/json",
            "Authorization": "64e5d1af081211d24e2cfecd",
            "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64; rv:147.0) Gecko/20100101 Firefox/147.0",
            "Accept": "*/*",
            "Accept-Language": "en-US,en;q=0.9",
            "Accept-Encoding": "gzip, deflate, br, zstd",
            "Referer": "https://saic.malaga.eu/",
            "Origin": "https://saic.malaga.eu",
            "Sec-GPC": "1",
            "Connection": "keep-alive",
            "Sec-Fetch-Dest": "empty",
            "Sec-Fetch-Mode": "cors",
            "Sec-Fetch-Site": "cross-site",
            "Priority": "u=4",
            "Pragma": "no-cache",
            "Cache-Control": "no-cache"
        }
    }
}

# Function to send a message
def send_message(user_msg):
    url = connector_config["base_url"] + connector_config["send_message"]["path"]
    headers = connector_config["send_message"]["headers"]
    
    payload = {
        "conversation": "6977595c744ccf01954b8797",
        "sender_type": "User",
        "sender": "6977595c3e4824018e496b0c",
        "bot": "64e5d1af081211d24e2cfec8",
        "language": "es",
        "url": "https://saic.malaga.eu/",
        "message": {
            "text": user_msg
        }
    }
    
    response = requests.post(url, headers=headers, json=payload)
    return response.json()

# Sending multiple requests
messages = ["hola", "¿cómo estás?", "¿qué puedes hacer?"]
for msg in messages:
    response = send_message(msg)
    print(f"Response for '{msg}': {response}")
```
