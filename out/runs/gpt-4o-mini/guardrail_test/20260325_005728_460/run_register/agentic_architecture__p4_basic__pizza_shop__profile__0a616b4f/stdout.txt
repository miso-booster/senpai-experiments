import openai

# Function to generate profile content using LLM
def generate_profile_content(prompt):
    response = openai.ChatCompletion.create(
        model="gpt-3.5-turbo",
        messages=[{"role": "user", "content": prompt}]
    )
    return response['choices'][0]['message']['content']

# Function to create a valid SENSEI profile
def create_sensei_profile(profile_name, user_simulation, chatbot_contract):
    # Define the profile structure
    profile = {
        "test_name": profile_name,

        "chatbot_contract": chatbot_contract,
        "conversation_controls": {
            "number_of_conversations": 5,
            "stopping_logic": "stop after 5 turns"
        }
    }
    
    # Convert to YAML
    profile_yaml = yaml.dump(profile, default_flow_style=False)
    
    # Save to file
    with open(f"{profile_name}.yaml", "w") as file:
        file.write(profile_yaml)
