Now, let's create a Python script that sends multiple requests to this connector. Here’s an example script:

```python
import requests
import yaml

# Load the YAML configuration
with open('ada-uam-custom.yaml', 'r') as file:
    config = yaml.safe_load(file)

# Prepare the base URL and headers
base_url = config['base_url'] + config['send_message']['path']
headers = config['send_message']['headers']

# Function to send a message to the chatbot
def send_message(user_msg):
    payload = config['send_message']['payload_template']
    payload['message']['text'] = user_msg  # Update the message text
    response = requests.post(base_url, headers=headers, json=payload)
    
    if response.status_code == 200:
        response_data = response.json()
        return response_data[config['response_path'].replace('.', '][').replace('text', 'text')]  # Extract the response text
    else:
        return f"Error: {response.status_code} - {response.text}"

# Example usage: Sending multiple requests
messages = ["hola", "¿cómo estás?", "¿qué tal tu día?"]
for msg in messages:
    reply = send_message(msg)
    print(f"User: {msg}\nBot: {reply}\n")
```
