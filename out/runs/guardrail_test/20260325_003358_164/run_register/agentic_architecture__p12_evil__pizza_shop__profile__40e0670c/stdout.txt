```python
import requests
import json

# Configuration for the connector
connector_config = {
    "base_url": "https://api.1millionbot.com",
    "headers": {
        "Content-Type": "application/json",
        "Authorization": "64e5d1af081211d24e2cfecd"
    }
}

# Function to send a message
def send_message(conversation_id, sender_id, bot_id, user_msg):
    url = f"{connector_config['base_url']}/api/public/messages"
    payload = {
        "conversation": conversation_id,
        "sender_type": "User",
        "sender": sender_id,
        "bot": bot_id,
        "language": "es",
        "url": "https://saic.malaga.eu/",
        "message": {
            "text": user_msg
        }
    }
    
    response = requests.post(url, headers=connector_config['headers'], json=payload)
    
    if response.status_code == 200:
        return response.json()
    else:
        return {"error": response.status_code, "message": response.text}

# Example usage: Sending multiple messages
conversation_id = "6977595c744ccf01954b8797"

bot_id = "64e5d1af081211d24e2cfec8"

for msg in messages:
    response = send_message(conversation_id, sender_id, bot_id, msg)
    print(json.dumps(response, indent=2, ensure_ascii=False))
```

- **requests**: A library to make HTTP requests.
- **send_message**: A function that constructs the request and sends it to the API.
