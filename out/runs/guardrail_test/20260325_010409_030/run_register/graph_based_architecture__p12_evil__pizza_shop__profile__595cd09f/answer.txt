Now, let's create a Python script that uses this connector to send multiple requests. Here's a simple example:

```python
import requests
import json

# Define the connector configuration
connector_config = {
    "base_url": "https://api.1millionbot.com",
    "headers": {
        "Content-Type": "application/json",
        "Authorization": "64e5d1af081211d24e2cfecd",
        "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64; rv:147.0) Gecko/20100101 Firefox/147.0",
        "Accept": "*/*",
        "Accept-Language": "en-US,en;q=0.9",
        "Accept-Encoding": "gzip, deflate, br, zstd",
        "Referer": "https://saic.malaga.eu/",
        "Origin": "https://saic.malaga.eu",
        "Sec-GPC": "1",
        "Connection": "keep-alive",
        "Sec-Fetch-Dest": "empty",
        "Sec-Fetch-Mode": "cors",
        "Sec-Fetch-Site": "cross-site",
        "Priority": "u=4",
        "Pragma": "no-cache",
        "Cache-Control": "no-cache"
    },
    "payload_template": {
        "conversation": "6977595c744ccf01954b8797",
        "sender_type": "User",
        "sender": "6977595c3e4824018e496b0c",
        "bot": "64e5d1af081211d24e2cfec8",
        "language": "es",
        "url": "https://saic.malaga.eu/",
        "message": {
            "text": ""
        }
    }
}

# Function to send a request
def send_request(user_message):
    payload = connector_config["payload_template"]
    payload["message"]["text"] = user_message

    response = requests.post(
        f"{connector_config['base_url']}/api/public/messages",
        headers=connector_config["headers"],
        data=json.dumps(payload)
    )

    return response.json()

# List of messages to send
messages = ["hola", "¿cómo estás?", "¿qué tal tu día?"]

# Sending multiple requests
for msg in messages:
    response = send_request(msg)
    print(f"Response for '{msg}': {response['response'][0]['text']}")
```
