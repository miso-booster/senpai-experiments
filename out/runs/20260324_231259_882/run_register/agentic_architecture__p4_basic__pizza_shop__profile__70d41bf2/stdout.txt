import requests
import yaml
from transformers import pipeline

# Step 1: Set up the LLM model
llm = pipeline("text-generation", model="gpt-3.5-turbo")  # Example model

# Step 2: Define a function to generate profile content
def generate_profile_content():
    prompt = "Create a SENSEI profile for a chatbot that assists users with pizza orders."
    response = llm(prompt, max_length=200)
    return response[0]['generated_text']

# Step 3: Format the content into a YAML structure
def create_profile_yaml(profile_content):
    profile = {
        "test_name": "PizzaOrderProfile",
        "llm_settings": {},
        "simulated_user": {
            "language": "en",
            "role": "customer",
            "context": "Ordering pizza",

        },
        "chatbot_contract": {
            "who_starts": "user",
            "fallback_message": "I'm sorry, I didn't understand that.",
            "structured_output_fields": ["size", "toppings", "price"]
        },
        "conversation_controls": {
            "number_of_conversations": 5,
            "stopping_logic": "user says goodbye"
        }
    }
    return yaml.dump(profile)

# Step 4: Save the profile to a YAML file
def save_profile_to_file(yaml_content, filename):
    with open(filename, 'w') as file:
        file.write(yaml_content)

# Main execution
if __name__ == "__main__":
    profile_content = generate_profile_content()
