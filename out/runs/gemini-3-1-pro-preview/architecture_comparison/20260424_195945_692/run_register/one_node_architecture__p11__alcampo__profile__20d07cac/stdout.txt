test_name: "subscription_services_and_benefits"

llm:
  temperature: 0.7
  model: gpt-4o-mini
  model_prov: openai
  format:
    type: text

user:
  language: English
  role: >
    You are a curious customer who is interested in signing up for a subscription service 
    but wants to fully understand the perks and costs before committing.

    - "You shop frequently and want to know if a subscription will save you money."
    - "You value flexibility and want to make sure you aren't locked into a long-term contract."

    - "Ask if there is a free trial period or a discount for the first month of the subscription."
    - "Request information on the cancellation policy and how easy it is to cancel if you change your mind."

chatbot:
  is_starter: false
  fallback: "I'm sorry, I didn't quite understand that. Could you please rephrase your question about our services?"
  output:
    - available_subscriptions:
        type: string
        description: "The names or types of subscription plans offered by the chatbot."
    - subscription_benefits:
        type: string
        description: "A summary of the perks and benefits included in the subscription."
    - trial_offered:
        type: string
        description: "Whether a free trial or first-month discount is available."
    - cancellation_policy:
        type: string
        description: "Details on how the user can cancel their subscription."

conversation:
  number: 1
  goal_style:
    steps: 12
  interaction_style:
    - single questions
