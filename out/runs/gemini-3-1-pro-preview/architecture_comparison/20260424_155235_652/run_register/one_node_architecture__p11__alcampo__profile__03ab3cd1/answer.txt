```yaml
test_name: "subscription_services_inquiry"

llm:
  temperature: 0.7
  model: gpt-4o
  format:
    type: text

user:
  language: English
  role: "A potential customer interested in learning about subscription services and their benefits."
  context:
    - "You want to know if there is a premium membership that offers free delivery or exclusive discounts."
    - "You are comparing different tiers of subscriptions to see which one fits your needs best."

    - "Ask the chatbot what subscription services or memberships are currently available."
    - "Inquire about the specific benefits and perks included in the premium subscription."
    - "Ask how much the monthly or annual fee is for the subscription."
    - "Find out if there is a free trial period available for new members."
    - "Ask how to cancel the subscription if you change your mind later."

  is_starter: false
  fallback: "I'm sorry, I didn't understand that. Could you please rephrase your question?"
  output:
    - available_subscriptions:
        type: string
        description: "Names of the subscription services offered."
    - subscription_benefits:
        type: string
        description: "List of perks and benefits for the subscriptions."
    - subscription_cost:
        type: string
        description: "The price or fee of the subscription services."
    - free_trial_available:
        type: string
        description: "Details on whether a free trial is offered."

  number: 1
  goal_style:
    steps: 15
```
