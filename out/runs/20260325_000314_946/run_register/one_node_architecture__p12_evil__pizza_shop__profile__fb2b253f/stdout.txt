```python
import requests
import json

# Define the connector configuration
connector_config = {
    "base_url": "https://api.1millionbot.com",
    "path": "/api/public/messages",
    "headers": {
        "Content-Type": "application/json",
        "Authorization": "64e5d1af081211d24e2cfecd",
        "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64; rv:147.0) Gecko/20100101 Firefox/147.0",
        "Accept": "*/*",
        "Accept-Language": "en-US,en;q=0.9",
        "Accept-Encoding": "gzip, deflate, br, zstd",
        "Referer": "https://saic.malaga.eu/",
        "Origin": "https://saic.malaga.eu",
        "Sec-GPC": "1",
        "Connection": "keep-alive",
        "Sec-Fetch-Dest": "empty",
        "Sec-Fetch-Mode": "cors",
        "Sec-Fetch-Site": "cross-site",
        "Priority": "u=4",
        "Pragma": "no-cache",
        "Cache-Control": "no-cache"
    }
}

# Function to send a message
def send_message(conversation_id, sender_id, bot_id, user_message):
    url = f"{connector_config['base_url']}{connector_config['path']}"
    payload = {
        "conversation": conversation_id,
        "sender_type": "User",
        "sender": sender_id,
        "bot": bot_id,
        "language": "es",
        "url": "https://saic.malaga.eu/",
        "message": {
            "text": user_message
        }
    }
    
    response = requests.post(url, headers=connector_config['headers'], json=payload)
    
    if response.status_code == 200:
        return response.json()
    else:
        return {"error": response.status_code, "message": response.text}

# Sending multiple requests
conversations = [
    {"conversation_id": "6977595c744ccf01954b8797", "sender_id": "6977595c3e4824018e496b0c", "bot_id": "64e5d1af081211d24e2cfec8", "user_message": "hola"},
    {"conversation_id": "6977595c744ccf01954b8798", "sender_id": "6977595c3e4824018e496b0c", "bot_id": "64e5d1af081211d24e2cfec8", "user_message": "¿Cómo estás?"},
    {"conversation_id": "6977595c744ccf01954b8799", "sender_id": "6977595c3e4824018e496b0c", "bot_id": "64e5d1af081211d24e2cfec8", "user_message": "¿Qué puedes hacer por mí?"}
]

# Loop through each conversation and send the message
for conv in conversations:
    response = send_message(conv["conversation_id"], conv["sender_id"], conv["bot_id"], conv["user_message"])
    print(json.dumps(response, indent=2, ensure_ascii=False))
```
