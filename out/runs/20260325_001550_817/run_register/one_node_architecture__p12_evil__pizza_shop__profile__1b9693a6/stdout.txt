import requests

connector_config = {

        "Sec-GPC": "1",
        "Connection": "keep-alive",
        "Sec-Fetch-Dest": "empty",
        "Sec-Fetch-Mode": "cors",
        "Sec-Fetch-Site": "cross-site",
        "Priority": "u=4",

        "Cache-Control": "no-cache"
    }
}

# Function to send a request
def send_request(conversation_id, sender_id, bot_id, user_message):
    url = connector_config["base_url"] + connector_config["path"]
    payload = {
        "conversation": conversation_id,
        "sender_type": "User",
        "sender": sender_id,
        "bot": bot_id,
        "language": "es",
        "url": "https://saic.malaga.eu/",
        "message": {
            "text": user_message
        }
    }
    
    response = requests.post(url, headers=connector_config["headers"], json=payload)
    return response.json()

# Example usage: sending multiple requests
if __name__ == "__main__":

    messages = ["hola", "¿cómo estás?", "¿qué puedes hacer?"]
    
    for message in messages:
        response = send_request(conversation_id, sender_id, bot_id, message)
        print(json.dumps(response, indent=4, ensure_ascii=False))
