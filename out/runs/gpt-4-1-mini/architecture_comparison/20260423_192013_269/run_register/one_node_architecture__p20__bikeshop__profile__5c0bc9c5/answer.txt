```yaml
test_name: "bike_wheel_change_appointment"

llm:
  temperature: 0.6
  model: gpt-4o-mini
  model_prov: openai
  format:
    type: text

user:
  language: English
  role: >
    You want to set an appointment at the bike shop to have your wheels changed.
    You will ask for available dates and times and confirm the appointment.
  context:
    - "You want the appointment to be scheduled as soon as possible."
    - "You might ask about available slots or propose specific dates and times."
  goals:
    - "I want to schedule an appointment for a wheel change on {{appointment_date}} at {{appointment_time}}."
    - appointment_date:
        function: forward()
        type: string
        data:
          - "Tomorrow"
          - "November 6th, 2024"
          - "Next Monday"
          - "Next Friday"
          - "December 1st, 2024"
    - appointment_time:
        function: another()
        type: string
        data:
          - "10:00 AM"
          - "2:00 PM"
          - "4:30 PM"
          - "9:00 AM"
          - "1:15 PM"

chatbot:
  is_starter: false
  fallback: "I'm sorry, I didn't catch that. Could you please repeat your request?"
  output:
    - confirmed_date:
        type: string
        description: "The date confirmed by the chatbot for the appointment."
    - confirmed_time:
        type: string
        description: "The time confirmed by the chatbot for the appointment."
    - appointment_confirmation:
        type: string
        description: "Full confirmation message from the chatbot about the appointment."

conversation:
  number: combinations
  goal_style:
    all_answered:
      export: true
      limit: 10
  interaction_style:
    - single questions
  max_cost: 0.5
```
