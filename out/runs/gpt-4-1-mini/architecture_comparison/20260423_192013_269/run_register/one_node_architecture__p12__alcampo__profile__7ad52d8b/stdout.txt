```yaml
  model_prov: openai
  format:
    type: text

user:
  language: English
  role: >
    You are a shopper who wants to add items to your online shopping cart.
    You ask the chatbot to add specific products with quantities to your cart.
  context:
    - "You want to add groceries and household items to your cart."
  goals:

        function: random(3)
        type: int

          - 1

          - 3
          - 4
          - 5
    - item_name:
        function: random(5)
        type: string
        data:
          - "milk"

          - "chicken breasts"
          - "rice"
          - "orange juice"

chatbot:
  is_starter: false
  fallback: "Sorry, I didn't understand that. Could you please specify the item and quantity?"
  output:
    - cart_items:
        type: list[string]
        description: "List of items added to the shopping cart with quantities"

conversation:
  number: 3
  goal_style:
    steps: 5
  interaction_style:
    - single questions
```
