```yaml
test_name: "bike_setup_and_customization"

llm:
  temperature: 0.6
  model: gpt-4o-mini
  model_prov: openai
  format:
    type: text

user:
  language: English
  role: >
    You recently bought a new bike and are visiting the bike shop to get it set up
    and customized. You want to change the seat and the handlebar grips to your liking,
    asking about options, prices, and scheduling the setup.

    - "You prefer ergonomic and durable components."
  goals:
    - "I would like to change the seat to a {{seat_type}}."
    - "I also want to replace the handlebar grips with {{grip_type}}."
    - "Can you tell me the price for the seat and grips replacement?"
    - "Please schedule an appointment for the bike setup."
    - seat_type:
        function: forward()
        type: string
        data: [comfort seat, racing seat, gel seat, leather seat]
    - grip_type:
        function: forward()
        type: string
        data: [foam grips, leather grips, ergonomic grips, rubber grips]

chatbot:
  is_starter: false
  fallback: "I'm sorry, I didn't catch that. Could you please repeat?"
  output:
    - seat_change_confirmation:
        type: string
        description: "Chatbot confirmation about changing the seat."
    - grip_change_confirmation:
        type: string
        description: "Chatbot confirmation about changing the handlebar grips."
    - seat_price:
        type: money
        description: "Price quoted for the seat replacement."
    - grip_price:
        type: money
        description: "Price quoted for the handlebar grips replacement."
    - appointment_confirmation:
        type: string
        description: "Confirmation message for the scheduled setup appointment."

conversation:
  number: 4
  goal_style:
    all_answered:
      export: true
      limit: 10
  interaction_style:
    - single questions
    - default
  max_cost: 0.7
```
