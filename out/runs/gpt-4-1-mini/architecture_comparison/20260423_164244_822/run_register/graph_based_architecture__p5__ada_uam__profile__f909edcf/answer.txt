  is_starter: false

        type: string
        description: "Correo electrónico del CAU"
    - physical_address:
        type: string
        description: "Ubicación física del CAU"

        type: string
        description: "Horario de atención del CAU"

  number: 3
  goal_style:
    steps: 10
  interaction_style:
    - single questions
