test_name: "bike_wheel_change_appointment"

llm:
  temperature: 0.6
  model: gpt-4o-mini
  model_prov: openai
  format:
    type: text

user:
  language: English
  role: >
    You are a customer wanting to schedule an appointment to have your bike wheels changed.
    You want to try different dates and times to see how the chatbot manages scheduling.
  context:
    - "You want to get your bike wheels replaced as soon as possible."
    - "You are flexible with dates and times but want to confirm the appointment."
  goals:
    - "I want to set an appointment for wheel change on {{appointment_date}} at {{appointment_time}}."
    - appointment_date:
        function: forward()
        type: string
        data:
          - "Tomorrow"
          - "Next Monday"
          - "November 10th, 2024"
          - "November 15th, 2024"
          - "Next Friday"
    - appointment_time:
        function: another()
        type: string
        data:
          - "10:00 AM"
          - "2:00 PM"
          - "4:30 PM"
          - "9:00 AM"
          - "1:00 PM"

chatbot:
  is_starter: false
  fallback: "I'm sorry, I didn't catch that. Could you please repeat your appointment request?"
  output:
    - confirmed_date:
        type: string
        description: "The date of the appointment confirmed by the chatbot."
    - confirmed_time:
        type: string
        description: "The time of the appointment confirmed by the chatbot."
    - appointment_confirmation:
        type: string
        description: "Full confirmation message from the chatbot regarding the appointment."

conversation:
  number: combinations
  goal_style:
    all_answered:
      export: true
      limit: 10
  interaction_style:
    - single questions
  max_cost: 0.5
