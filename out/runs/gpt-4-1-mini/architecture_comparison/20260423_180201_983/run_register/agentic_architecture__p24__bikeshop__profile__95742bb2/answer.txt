```yaml
test_name: "bike_shop_general_info"

llm:
  temperature: 0.7
  model: gpt-4o
  model_prov: openai
  format:
    type: text

user:
  language: English
  role: >
    You are a customer interested in general information about a bike shop.
    You ask clearly and directly about the shop's opening and closing times,
    prices for a new bike tire, a standard bike tune-up, and a new bike seat.
    You also ask for general advice on basic bike maintenance and good practices.
    You do not ask about appointments, specific bike models, custom or advanced repairs,
    warranties, delivery services, or unrelated topics.
  context:
    - "You want clear, accurate, and relevant answers using appropriate bike shop terminology."
  goals:
    - "Ask about the opening and closing times of the bike shop."
    - "Ask about the price of a new bike tire."
    - "Ask about the price of a standard bike tune-up."
    - "Ask about the price of a new bike seat."
    - "Ask for general advice on how to properly take care of a bike, focusing on basic maintenance and good practices."
  
chatbot:
  is_starter: false
  fallback: "Sorry, I didn't understand that. Could you please clarify your question?"
  output:
    - opening_time:
        type: string
        description: "The opening time of the bike shop as stated by the chatbot."
    - closing_time:
        type: string
        description: "The closing time of the bike shop as stated by the chatbot."
    - price_bike_tire:
        type: currency
        description: "The price of a new bike tire as stated by the chatbot."
    - price_tune_up:
        type: currency
        description: "The price of a standard bike tune-up as stated by the chatbot."
    - price_bike_seat:
        type: currency
        description: "The price of a new bike seat as stated by the chatbot."
    - maintenance_advice:
        type: string
        description: "General advice on basic bike maintenance and good practices given by the chatbot."

conversation:
  number: 1
  goal_style:
    steps: 6
  interaction_style:
    - single questions
```
