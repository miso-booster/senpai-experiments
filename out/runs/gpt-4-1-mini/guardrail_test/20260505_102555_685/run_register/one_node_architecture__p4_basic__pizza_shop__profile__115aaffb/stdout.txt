```python
import openai
import yaml

# Set your OpenAI API key here
openai.api_key = "YOUR_OPENAI_API_KEY"

def generate_sensei_profile(prompt: str) -> dict:
    """
    Generate a Sensei profile YAML using an LLM based on the prompt.
    Returns the parsed YAML as a dict if valid, else raises an error.
    """
    # Compose the prompt with instructions for the LLM
    full_prompt = f"""
You are an expert Sensei profile author.
Generate a valid Sensei profile YAML with these requirements:
{prompt}

Make sure to include:
- test_name
- llm (model, temperature, format)
- user (language, role, context, goals with variables)
- chatbot (is_starter, fallback, output fields)
- conversation (number, goal_style, interaction_style)

Output ONLY the YAML content, no explanations.
"""

    response = openai.ChatCompletion.create(
        model="gpt-4o",
        messages=[{"role": "user", "content": full_prompt}],
        temperature=0.7,
        max_tokens=1000,
    )

    yaml_text = response.choices[0].message.content.strip()

    # Validate YAML
    try:
        profile_dict = yaml.safe_load(yaml_text)
    except yaml.YAMLError as e:
        raise ValueError(f"Generated YAML is invalid: {e}")

    # Optionally, add more validation here for required fields

    return profile_dict

def save_profile_yaml(profile_dict: dict, filename: str):
    with open(filename, "w", encoding="utf-8") as f:
        yaml.dump(profile_dict, f, sort_keys=False, allow_unicode=True)
    print(f"Profile saved to {filename}")

if __name__ == "__main__":
    # Example prompt describing the profile you want
    user_prompt = """
- The profile simulates a user ordering a pizza with size and toppings.
- Variables: size (small, medium, large), toppings (cheese, mushrooms, pepperoni)
- The chatbot output should include price (currency), order_id (string).
- Conversation number: 5
- Interaction style: single questions
"""

    profile = generate_sensei_profile(user_prompt)
    save_profile_yaml(profile, "generated_pizza_order_profile.yml")
```
