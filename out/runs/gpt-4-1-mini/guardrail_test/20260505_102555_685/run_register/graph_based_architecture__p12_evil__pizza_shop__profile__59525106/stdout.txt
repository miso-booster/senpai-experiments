from chatbot_connectors.implementations.custom import CustomChatbot

def main():
    # Load the connector YAML file (assuming saved as 'saic-malaga-bot.yml')
    bot = CustomChatbot("saic-malaga-bot.yml")

    messages = [
        "hola",
        "¿Cuál es el horario de atención?",
        "Necesito ayuda con mi cuenta",
        "Gracias"
    ]

    for msg in messages:
        success, response = bot.execute_with_input(msg)
        if success:
            print(f"User: {msg}")
            print(f"Bot: {response}\n")
        else:
            print(f"Failed to get response for message: {msg}")

if __name__ == "__main__":
    main()
