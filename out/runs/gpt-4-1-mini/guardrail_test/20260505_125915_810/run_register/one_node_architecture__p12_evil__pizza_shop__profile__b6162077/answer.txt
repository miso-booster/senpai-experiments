And here is a complete Python script that uses this connector YAML to send multiple messages to the chatbot:

```python
import yaml
import requests

class CustomConnector:
    def __init__(self, config_path):
        with open(config_path, 'r', encoding='utf-8') as f:
            self.config = yaml.safe_load(f)
        self.base_url = self.config['base_url']
        self.path = self.config['send_message']['path']
        self.method = self.config['send_message'].get('method', 'POST').upper()
        self.headers = self.config['send_message'].get('headers', {})
        self.payload_template = self.config['send_message']['payload_template']
        self.response_path = self.config.get('response_path', '')

    def send_message(self, user_msg):
        # Fill in the user message in the payload
        payload = self._fill_payload(user_msg)
        url = self.base_url + self.path

        if self.method == 'POST':
            response = requests.post(url, json=payload, headers=self.headers)
        else:
            raise NotImplementedError(f"HTTP method {self.method} not implemented")

        response.raise_for_status()
        data = response.json()
        return self._extract_response(data)

    def _fill_payload(self, user_msg):
        # Recursively replace '{user_msg}' in the payload template
        def replace(obj):
            if isinstance(obj, dict):
                return {k: replace(v) for k, v in obj.items()}
            elif isinstance(obj, list):
                return [replace(i) for i in obj]
            elif isinstance(obj, str):
                return obj.replace("{user_msg}", user_msg)
            else:
                return obj
        return replace(self.payload_template)

    def _extract_response(self, data):
        # Extract response using dot notation path
        parts = self.response_path.split('.')
        current = data
        for part in parts:
            if part.isdigit():
                current = current[int(part)]
            else:
                current = current.get(part, None)
            if current is None:
                break
        return current

if __name__ == "__main__":
    connector = CustomConnector("connector.yaml")  # path to the YAML file above

    messages = [
        "hola",
        "¿Cuál es el horario de atención?",
        "¿Pueden ayudarme con mi pedido?",
        "Gracias, adiós."
    ]

    for msg in messages:
        try:
            reply = connector.send_message(msg)
            print(f"User: {msg}")
            print(f"Bot: {reply}\n")
        except Exception as e:
            print(f"Failed to send message '{msg}': {e}")
```
