import openai
import yaml

openai.api_key = "YOUR_OPENAI_API_KEY"

def generate_sensei_profile(prompt):
    system_prompt = """
    You are an assistant that generates valid Sensei user profiles in YAML format.
    The profile must include fields like test_name, llm, user, chatbot, and conversation.
    The output must be valid YAML suitable for Sensei.
    """
    response = openai.ChatCompletion.create(
        model="gpt-4o",
        messages=[
            {"role": "system", "content": system_prompt},
            {"role": "user", "content": prompt}
        ],
        temperature=0.7,
        max_tokens=500
    )
    yaml_text = response['choices'][0]['message']['content']
    try:
        profile = yaml.safe_load(yaml_text)
        print("Generated valid Sensei profile YAML:")
        print(yaml.dump(profile, sort_keys=False))
        return profile
    except yaml.YAMLError as e:
        print("Error parsing YAML:", e)
        print("Raw output was:\n", yaml_text)
        return None

if __name__ == "__main__":
    user_prompt = """
    Create a Sensei profile for a user ordering a pizza with variables for size (small, medium, large),

    for price (currency), order_id (string), and estimated_time (time).
