Here's a complete Python script that uses the connector details to send multiple requests to the chatbot API:

```python
import requests

class MillionBotConnector:
    def __init__(self):
        self.base_url = "https://api.1millionbot.com"
        self.path = "/api/public/messages"
        self.headers = {
            "Content-Type": "application/json",
            "Authorization": "64e5d1af081211d24e2cfecd"
        }
        self.fixed_payload = {
            "conversation": "6977595c744ccf01954b8797",
            "sender_type": "User",
            "sender": "6977595c3e4824018e496b0c",
            "bot": "64e5d1af081211d24e2cfec8",
            "language": "es",
            "url": "https://saic.malaga.eu/"
        }

    def send_message(self, user_msg):
        payload = self.fixed_payload.copy()
        payload["message"] = {"text": user_msg}
        response = requests.post(
            self.base_url + self.path,
            json=payload,
            headers=self.headers
        )
        if response.status_code == 200:
            data = response.json()
            # Extract bot reply from response.0.text
            try:
                bot_reply = data["response"][0]["text"]
                return bot_reply
            except (KeyError, IndexError):
                return "Error: Unexpected response format."
        else:
            return f"Error: HTTP {response.status_code}"

if __name__ == "__main__":
    bot = MillionBotConnector()
    user_messages = [
        "hola",
        "¿Cuál es el horario de atención?",
        "¿Cómo puedo contactar con soporte?",
    ]

    for msg in user_messages:
        print(f"User: {msg}")
        reply = bot.send_message(msg)
        print(f"Bot: {reply}\n")
```

This script:
- Defines a connector class with the fixed headers and payload fields.
- Sends POST requests with the user message dynamically inserted.
- Extracts and prints the bot's reply from the JSON response.
- Sends multiple example messages in a loop.
