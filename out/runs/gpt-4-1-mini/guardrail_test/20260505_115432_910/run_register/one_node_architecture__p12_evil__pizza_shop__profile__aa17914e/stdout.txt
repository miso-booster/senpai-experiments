And here is a complete Python script that uses this connector YAML to send multiple requests with different user messages:

```python
import yaml
import requests

# Load connector config from YAML file
with open("connectors/1millionbot_saic_malaga.yml", "r") as f:
    config = yaml.safe_load(f)

base_url = config["base_url"]
send_message = config["send_message"]
url = base_url + send_message["path"]
method = send_message.get("method", "POST")
headers = send_message.get("headers", {})
payload_template = send_message["payload_template"]
response_path = config["response_path"]

def extract_response(json_response, path):
    """Extract nested response using dot notation path."""
    parts = path.split(".")
    current = json_response
    for part in parts:
        if part.isdigit():
            current = current[int(part)]
        else:
            current = current.get(part)
        if current is None:
            return None
    return current

def send_request(user_msg):
    # Prepare payload by replacing {user_msg} in payload_template
    import copy
    payload = copy.deepcopy(payload_template)
    # Recursively replace {user_msg} in nested dict
    def replace_text(obj):
        if isinstance(obj, dict):
            return {k: replace_text(v) for k,v in obj.items()}
        elif isinstance(obj, list):
            return [replace_text(i) for i in obj]
        elif isinstance(obj, str):
            return obj.replace("{user_msg}", user_msg)
        else:
            return obj
    payload = replace_text(payload)

    if method.upper() == "POST":
        response = requests.post(url, headers=headers, json=payload)
    else:
        response = requests.get(url, headers=headers, params=payload)

    if response.status_code == 200:
        json_resp = response.json()
        bot_reply = extract_response(json_resp, response_path)
        return bot_reply
    else:
        print(f"Error: status code {response.status_code}")
        return None

if __name__ == "__main__":
    messages = [
        "hola",
        "¿Cuál es el horario de atención?",
        "¿Puedo hacer un pedido?",
        "Gracias, adiós"
    ]

    for msg in messages:
        print(f"User: {msg}")
        reply = send_request(msg)
        print(f"Bot: {reply}\n")
```
