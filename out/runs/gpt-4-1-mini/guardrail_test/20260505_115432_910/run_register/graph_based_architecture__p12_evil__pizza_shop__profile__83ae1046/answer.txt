### Now, here is a complete Python script to send multiple requests using this connector:

```python
import requests
import json

class SaicMalagaBotConnector:
    def __init__(self):
        self.base_url = "https://api.1millionbot.com"
        self.path = "/api/public/messages"
        self.headers = {
            "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64; rv:147.0) Gecko/20100101 Firefox/147.0",
            "Accept": "*/*",
            "Accept-Language": "en-US,en;q=0.9",
            "Accept-Encoding": "gzip, deflate, br, zstd",
            "Referer": "https://saic.malaga.eu/",
            "Content-Type": "application/json",
            "Authorization": "64e5d1af081211d24e2cfecd",
            "Origin": "https://saic.malaga.eu",
            "Sec-GPC": "1",
            "Connection": "keep-alive",
            "Sec-Fetch-Dest": "empty",
            "Sec-Fetch-Mode": "cors",
            "Sec-Fetch-Site": "cross-site",
            "Priority": "u=4",
            "Pragma": "no-cache",
            "Cache-Control": "no-cache"
        }
        self.payload_template = {
            "conversation": "6977595c744ccf01954b8797",
            "sender_type": "User",
            "sender": "6977595c3e4824018e496b0c",
            "bot": "64e5d1af081211d24e2cfec8",
            "language": "es",
            "url": "https://saic.malaga.eu/",
            "message": {
                "text": ""
            }
        }

    def send_message(self, user_msg):
        self.payload_template["message"]["text"] = user_msg
        url = self.base_url + self.path
        response = requests.post(url, headers=self.headers, json=self.payload_template)
        if response.status_code == 200:
            try:
                data = response.json()
                # Extract response text using the response path: response[0].text
                bot_reply = data.get("response", [{}])[0].get("text", "")
                return bot_reply
            except json.JSONDecodeError:
                return "Error decoding JSON response"
        else:
            return f"Request failed with status code {response.status_code}"

if __name__ == "__main__":
    bot = SaicMalagaBotConnector()
    messages = [
        "hola",
        "¿Cómo estás?",
        "¿Qué servicios ofrecen?",
        "Gracias"
    ]

    for msg in messages:
        reply = bot.send_message(msg)
        print(f"User: {msg}")
        print(f"Bot: {reply}\n")
```

---

### How this script works:

- Defines a class `SaicMalagaBotConnector` with your API details.
- The `send_message` method sends a POST request with the user message.
- It parses the JSON response and extracts the bot's reply from `response[0].text`.
- The main block sends multiple messages and prints both user and bot messages.
