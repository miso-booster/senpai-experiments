- Profiles are YAML files defining simulated user behavior for chatbot testing.
- They include sections like `test_name`, `llm` settings, `user` (language, role, context, goals), `chatbot` (starter flag, fallback, output schema), and `conversation` controls.
- The `chatbot.output` section defines fields extracted from conversations, which rules later validate.
- Variables in `user.goals` must be declared with types and data sources.
- Interaction styles and conversation controls influence the simulation.
- Profiles must be valid YAML and follow the documented schema.

import yaml
import openai  # or other LLM API client

def generate_sensei_profile(prompt_details: dict, openai_api_key: str) -> str:
    """
    Generate a valid SENSEI profile YAML using an LLM.
    
    prompt_details: dict with keys like 'test_name', 'user_role', 'goals', 'output_fields', 'conversation_params'.
    openai_api_key: API key for OpenAI or other LLM provider.
    
    Returns: YAML string of the profile.
    """
    openai.api_key = openai_api_key
    
    # Compose prompt for LLM
    system_prompt = """
    You are an expert in SENSEI chatbot testing profiles.
    Generate a valid SENSEI profile YAML that matches the given details.
    Follow this structure strictly:
    - test_name: string
    - llm: temperature, model, model_prov, format
    - user: language, role, context, goals (with variables and functions)
    - chatbot: is_starter, fallback, output (fields with type and description)
    - conversation: number, goal_style, interaction_style
    
    Use the following example as a template:
    test_name: "example_profile"
    llm:
      temperature: 0.7
      model: gpt-4o
      model_prov: openai
      format:
        type: text
    user:
      language: English
      role: "Simulated user role description"
      context:
        - "Context details"
      goals:
        - "Goal with {{variable}}"
        - variable:
            function: random()
            type: string
            data: [option1, option2]
    chatbot:
      is_starter: false
      fallback: "Sorry, can you rephrase?"
      output:
        - field1:
            type: string
            description: "Description of field1"
    conversation:
      number: 2
      goal_style:
        steps: 5
      interaction_style:
        - random:
          - long phrases
          - make spelling mistakes
    """
    
    user_prompt = f"""
    Generate a SENSEI profile YAML with these details:
    Test name: {prompt_details['test_name']}
    User role: {prompt_details['user_role']}
    User language: {prompt_details.get('user_language', 'English')}
    Goals: {prompt_details['goals']}
    Chatbot output fields: {prompt_details['output_fields']}
    Conversation parameters: {prompt_details.get('conversation_params', {})}
    """
    
    response = openai.ChatCompletion.create(
        model="gpt-4o",
        messages=[
            {"role": "system", "content": system_prompt},
            {"role": "user", "content": user_prompt}
        ],
        temperature=0.7,
        max_tokens=1000,
    )
    
    profile_yaml = response['choices'][0]['message']['content']
    
    # Validate YAML syntax
    try:
        profile_data = yaml.safe_load(profile_yaml)
    except yaml.YAMLError as e:
        raise ValueError(f"Generated YAML is invalid: {e}")
    
    # Additional validation can be added here
    
    return profile_yaml
