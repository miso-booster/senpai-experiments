```python
import openai
import yaml

openai.api_key = "YOUR_OPENAI_API_KEY"

def generate_sensei_profile(prompt):
    # Compose a prompt asking the LLM to generate a Sensei profile YAML
    full_prompt = f"""
    Generate a valid Sensei profile YAML for a chatbot user simulator.
    Follow the Sensei profile schema strictly, including:
    - test_name
    - llm (temperature, model, format)
    - user (language, role, context, goals with variables)
    - chatbot (is_starter, fallback, output fields)
    - conversation (number, goal_style, interaction_style)

    Profile description:
    {prompt}

    Return only the YAML content.
    """

    response = openai.ChatCompletion.create(
        model="gpt-4o",
        messages=[{"role": "user", "content": full_prompt}],
        temperature=0.7,
        max_tokens=800,
    )
    return response.choices[0].message.content

def save_profile_yaml(yaml_content, filename):
    with open(filename, "w", encoding="utf-8") as f:
        f.write(yaml_content)

if __name__ == "__main__":
    description = "A user ordering a pizza with size, toppings, and drink preferences."
    profile_yaml = generate_sensei_profile(description)
    print("Generated Profile YAML:\n", profile_yaml)

    save_profile_yaml(profile_yaml, "generated_pizza_order_profile.yml")
    print("Profile saved to generated_pizza_order_profile.yml")
```

### How this works:
- You provide a natural language description of the profile you want.
- The LLM generates a valid Sensei profile YAML string.
- You save it to a file for use in your Sensei project.
