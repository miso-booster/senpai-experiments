And here is a complete Python script that uses this connector configuration to send multiple messages sequentially:

```python
import requests
import yaml

# Load the connector YAML configuration
connector_yaml = """
name: "1MillionBot Custom"
base_url: "https://api.1millionbot.com"
send_message:
  path: "/api/public/messages"
  method: "POST"
  headers:
    User-Agent: "Mozilla/5.0 (Windows NT 10.0; Win64; x64; rv:147.0) Gecko/20100101 Firefox/147.0"
    Accept: "*/*"
    Accept-Language: "en-US,en;q=0.9"
    Accept-Encoding: "gzip, deflate, br, zstd"
    Referer: "https://saic.malaga.eu/"
    Content-Type: "application/json"
    Authorization: "64e5d1af081211d24e2cfecd"
    Origin: "https://saic.malaga.eu"
    Sec-GPC: "1"
    Connection: "keep-alive"
    Sec-Fetch-Dest: "empty"
    Sec-Fetch-Mode: "cors"
    Sec-Fetch-Site: "cross-site"
    Priority: "u=4"
    Pragma: "no-cache"
    Cache-Control: "no-cache"
  payload_template:
    conversation: "6977595c744ccf01954b8797"
    sender_type: "User"
    sender: "6977595c3e4824018e496b0c"
    bot: "64e5d1af081211d24e2cfec8"
    language: "es"
    url: "https://saic.malaga.eu/"
    message:
      text: "{user_msg}"
response_path: "response.0.text"
"""

config = yaml.safe_load(connector_yaml)

def send_message(user_msg):
    url = config["base_url"] + config["send_message"]["path"]
    headers = config["send_message"].get("headers", {})
    payload = config["send_message"]["payload_template"].copy()

    # Deep copy for nested dicts
    import copy
    payload = copy.deepcopy(payload)

    # Insert user message
    payload["message"]["text"] = user_msg

    response = requests.post(url, json=payload, headers=headers)
    response.raise_for_status()
    data = response.json()

    # Extract response text using response_path
    # The response_path is dot-separated keys/indexes
    path_parts = config["response_path"].split(".")
    current = data
    for part in path_parts:
        if part.isdigit():
            current = current[int(part)]
        else:
            current = current.get(part, None)
            if current is None:
                break
    return current

if __name__ == "__main__":
    messages = [
        "hola",
        "¿Cuál es el horario de atención?",
        "¿Qué pizzas tienen?",
        "¿Puedo hacer un pedido para recoger?",
        "Gracias, eso es todo."
    ]

    for msg in messages:
        try:
            reply = send_message(msg)
            print(f"User: {msg}")
            print(f"Bot: {reply}\n")
        except Exception as e:
            print(f"Error sending message '{msg}': {e}")
```
