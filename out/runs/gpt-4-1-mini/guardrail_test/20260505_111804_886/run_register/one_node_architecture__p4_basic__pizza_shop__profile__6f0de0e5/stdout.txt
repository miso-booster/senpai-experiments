```python
import openai
import yaml

openai.api_key = "your-openai-api-key"

def generate_sensei_profile(test_name, user_role, user_goals, chatbot_outputs, language="English"):
    prompt = f"""
You are an expert Sensei profile author. Generate a valid Sensei profile YAML for a chatbot test.

Requirements:
- test_name: {test_name}
- user.language: {language}
- user.role: {user_role}
- user.goals: {user_goals}
- chatbot.output: {chatbot_outputs}

Follow the Sensei Profiles specification strictly. Use appropriate types for chatbot outputs (string, int, currency, time, etc.).
Include a conversation section with number: 3, goal_style: steps: 5, and interaction_style: single question.

Return only the YAML content without extra explanations.
"""

    response = openai.ChatCompletion.create(
        model="gpt-4o-mini",
        messages=[{"role": "user", "content": prompt}],
        temperature=0.7,
        max_tokens=500,
    )

    profile_yaml = response.choices[0].message.content.strip()
    return profile_yaml

if __name__ == "__main__":
```
