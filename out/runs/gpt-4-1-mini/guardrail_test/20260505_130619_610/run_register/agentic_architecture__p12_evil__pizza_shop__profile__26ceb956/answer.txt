Here's a complete Python script to send multiple requests to the connector using the requests library. It sends different user messages and prints the bot's responses:

```python
import requests
import json

class MillionBotConnector:
    def __init__(self):
        self.url = "https://api.1millionbot.com/api/public/messages"
        self.headers = {
            "Content-Type": "application/json",
            "Authorization": "64e5d1af081211d24e2cfecd",
            "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64; rv:147.0) Gecko/20100101 Firefox/147.0",
            "Accept": "*/*",
            "Accept-Language": "en-US,en;q=0.9",
            "Accept-Encoding": "gzip, deflate, br, zstd",
            "Referer": "https://saic.malaga.eu/",
            "Origin": "https://saic.malaga.eu",
            "Sec-GPC": "1",
            "Connection": "keep-alive",
            "Sec-Fetch-Dest": "empty",
            "Sec-Fetch-Mode": "cors",
            "Sec-Fetch-Site": "cross-site",
            "Priority": "u=4",
            "Pragma": "no-cache",
            "Cache-Control": "no-cache"
        }
        self.payload_base = {
            "conversation": "6977595c744ccf01954b8797",
            "sender_type": "User",
            "sender": "6977595c3e4824018e496b0c",
            "bot": "64e5d1af081211d24e2cfec8",
            "language": "es",
            "url": "https://saic.malaga.eu/",
            "message": {
                "text": ""
            }
        }

    def send_message(self, user_msg):
        payload = self.payload_base.copy()
        payload["message"]["text"] = user_msg
        response = requests.post(self.url, headers=self.headers, json=payload)
        if response.status_code == 200:
            try:
                data = response.json()
                # Extract the bot response text from the expected path
                bot_reply = data["response"][0]["text"]
                return bot_reply
            except (KeyError, IndexError, json.JSONDecodeError):
                return "Error parsing response"
        else:
            return f"Request failed with status code {response.status_code}"

if __name__ == "__main__":
    bot = MillionBotConnector()
    user_messages = [
        "hola",
        "¿Cuál es el horario de atención?",
        "Necesito ayuda con mi cuenta",
        "Gracias, eso es todo"
    ]
    for msg in user_messages:
        reply = bot.send_message(msg)
        print(f"User: {msg}")
        print(f"Bot: {reply}\n")
```
