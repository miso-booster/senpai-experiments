And here's a complete Python script that uses this connector YAML to send multiple messages to the chatbot:

```python
import yaml
import requests

class CustomConnector:
    def __init__(self, config_path):
        with open(config_path, 'r', encoding='utf-8') as f:
            self.config = yaml.safe_load(f)
        self.base_url = self.config['base_url']
        self.path = self.config['send_message']['path']
        self.method = self.config['send_message'].get('method', 'POST').upper()
        self.headers = self.config['send_message'].get('headers', {})
        self.payload_template = self.config['send_message']['payload_template']
        self.response_path = self.config['response_path']

    def send_message(self, user_msg):
        payload = self._build_payload(user_msg)
        url = self.base_url + self.path
        if self.method == 'POST':
            response = requests.post(url, json=payload, headers=self.headers)
        else:
            response = requests.get(url, params=payload, headers=self.headers)
        response.raise_for_status()
        return self._extract_response(response.json())

    def _build_payload(self, user_msg):
        # Deep copy and replace {user_msg} in payload_template
        import copy
        payload = copy.deepcopy(self.payload_template)

        def replace_text(obj):
            if isinstance(obj, dict):
                return {k: replace_text(v) for k, v in obj.items()}
            elif isinstance(obj, list):
                return [replace_text(i) for i in obj]
            elif isinstance(obj, str):
                return obj.replace("{user_msg}", user_msg)
            else:
                return obj

        return replace_text(payload)

    def _extract_response(self, json_response):
        # Navigate the response_path like 'response.0.text'
        parts = self.response_path.split('.')
        current = json_response
        for part in parts:
            if part.isdigit():
                idx = int(part)
                current = current[idx]
            else:
                current = current.get(part)
            if current is None:
                break
        return current

if __name__ == "__main__":
    connector = CustomConnector("path_to_connector.yaml")  # Replace with your YAML file path

    messages = [
        "hola",
        "¿Cuál es el horario de atención?",
        "¿Qué pizzas ofrecen?",
        "Quiero hacer un pedido",
        "Gracias"
    ]

    for msg in messages:
        try:
            reply = connector.send_message(msg)
            print(f"User: {msg}")
            print(f"Bot: {reply}\n")
        except Exception as e:
            print(f"Error sending message '{msg}': {e}")
```
