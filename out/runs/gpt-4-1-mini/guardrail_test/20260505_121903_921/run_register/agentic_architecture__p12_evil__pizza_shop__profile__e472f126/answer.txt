```python
import requests
import json

class OneMillionBotConnector:
    def __init__(self):
        self.base_url = "https://api.1millionbot.com"
        self.path = "/api/public/messages"
        self.headers = {
            "Content-Type": "application/json",
            "Authorization": "64e5d1af081211d24e2cfecd"
        }
        self.payload_template = {
            "conversation": "6977595c744ccf01954b8797",
            "sender_type": "User",
            "sender": "6977595c3e4824018e496b0c",
            "bot": "64e5d1af081211d24e2cfec8",
            "language": "es",
            "url": "https://saic.malaga.eu/",
            "message": {
                "text": ""
            }
        }

    def send_message(self, user_msg):
        payload = self.payload_template.copy()
        payload["message"]["text"] = user_msg
        response = requests.post(
            self.base_url + self.path,
            headers=self.headers,
            data=json.dumps(payload)
        )
        if response.status_code == 200:
            try:
                data = response.json()
                # Extract response text from response.0.text path
                return data["response"][0]["text"]
            except (KeyError, IndexError, json.JSONDecodeError):
                return "Error parsing response"
        else:
            return f"Request failed with status {response.status_code}"

if __name__ == "__main__":
    bot = OneMillionBotConnector()
    messages = [
        "hola",
        "¿Cómo estás?",
        "¿Qué puedes hacer?",
        "Gracias"
    ]
    for msg in messages:
        reply = bot.send_message(msg)
        print(f"User: {msg}")
        print(f"Bot: {reply}\n")
```
