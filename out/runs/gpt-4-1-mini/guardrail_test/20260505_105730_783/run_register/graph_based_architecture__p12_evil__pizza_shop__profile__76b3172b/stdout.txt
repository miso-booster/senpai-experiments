### Python Script to Send Multiple Requests

from chatbot_connectors.implementations.custom import CustomChatbot

def main():
    # Path to your YAML connector file
    connector_path = "path/to/your/1millionbot-malaga.yaml"
    
    # Initialize the chatbot with the connector config
    bot = CustomChatbot(connector_path)
    
    # List of messages to send
    messages = [
        "hola",
        "¿Cómo estás?",
        "¿Qué servicios ofrecen?",
        "Gracias"
    ]
    
    for msg in messages:
        success, response = bot.execute_with_input(msg)
        if success:
            print(f"User: {msg}")
            print(f"Bot: {response}\n")
        else:
            print(f"Failed to get response for message: {msg}")

if __name__ == "__main__":
    main()
