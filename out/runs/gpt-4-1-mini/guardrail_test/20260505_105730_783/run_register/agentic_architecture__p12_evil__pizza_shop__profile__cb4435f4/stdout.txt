Here is a complete Python script that uses the `requests` library to send multiple messages to the MillionBot API according to the connector configuration:

```python
import requests

class MillionBotConnector:
    def __init__(self):
        self.base_url = "https://api.1millionbot.com"
        self.path = "/api/public/messages"
        self.headers = {
            "Content-Type": "application/json",
            "Authorization": "64e5d1af081211d24e2cfecd"
        }
        self.payload_template = {
            "conversation": "6977595c744ccf01954b8797",
            "sender_type": "User",
            "sender": "6977595c3e4824018e496b0c",
            "bot": "64e5d1af081211d24e2cfec8",
            "language": "es",
            "url": "https://saic.malaga.eu/",
            "message": {
                "text": ""
            }
        }

    def send_message(self, user_msg):
        payload = self.payload_template.copy()
        payload["message"]["text"] = user_msg
        response = requests.post(self.base_url + self.path, json=payload, headers=self.headers)
        if response.status_code == 200:
            try:
                data = response.json()
                # Extract the bot reply text from response.0.text
                bot_reply = data.get("response", [{}])[0].get("text", "")
                return bot_reply
            except Exception as e:
                return f"Error parsing response JSON: {e}"
        else:
            return f"Request failed with status code {response.status_code}"

if __name__ == "__main__":
    bot = MillionBotConnector()
    messages = [
        "hola",
        "¿Cómo estás?",
        "¿Puedes ayudarme con información sobre Málaga?",
        "Gracias"
    ]

    for msg in messages:
        reply = bot.send_message(msg)
        print(f"User: {msg}")
        print(f"Bot: {reply}")
        print("-" * 40)
```

This script:
- Defines a connector class with the base URL, headers, and payload template from the YAML.
- Sends multiple user messages in sequence.
- Prints the bot's response extracted from the JSON response path `response.0.text`.
