And here is a complete Python script that uses this connector YAML to send multiple messages to the chatbot and print the responses:

```python
import yaml
import requests

class CustomConnector:
    def __init__(self, config_path):
        with open(config_path, 'r', encoding='utf-8') as f:
            self.config = yaml.safe_load(f)
        self.base_url = self.config['base_url']
        self.path = self.config['send_message']['path']
        self.method = self.config['send_message'].get('method', 'POST').upper()
        self.headers = self.config['send_message'].get('headers', {})
        self.payload_template = self.config['send_message']['payload_template']
        self.response_path = self.config['response_path']

    def send_message(self, user_msg):
        # Prepare payload by replacing {user_msg} in payload_template
        def replace_placeholders(obj):
            if isinstance(obj, dict):
                return {k: replace_placeholders(v) for k, v in obj.items()}
            elif isinstance(obj, list):
                return [replace_placeholders(i) for i in obj]
            elif isinstance(obj, str):
                return obj.replace("{user_msg}", user_msg)
            else:
                return obj

        payload = replace_placeholders(self.payload_template)

        url = self.base_url + self.path
        if self.method == 'POST':
            response = requests.post(url, json=payload, headers=self.headers)
        elif self.method == 'GET':
            response = requests.get(url, params=payload, headers=self.headers)
        else:
            raise ValueError(f"Unsupported HTTP method: {self.method}")

        response.raise_for_status()
        data = response.json()

        # Extract response using dot notation path
        parts = self.response_path.split('.')
        result = data
        for part in parts:
            if part.isdigit():
                result = result[int(part)]
            else:
                result = result.get(part)
            if result is None:
                break
        return result

if __name__ == "__main__":
    connector = CustomConnector("connector.yaml")  # Save the YAML above as connector.yaml

    messages = [
        "hola",
        "¿Cuál es el horario de atención?",
        "¿Pueden ayudarme con mi pedido?",
        "Gracias, eso es todo."
    ]

    for msg in messages:
        try:
            reply = connector.send_message(msg)
            print(f"User: {msg}")
            print(f"Bot: {reply}\n")
        except Exception as e:
            print(f"Error sending message '{msg}': {e}")
```
