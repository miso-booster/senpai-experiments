Here's a complete Python script to send multiple requests to the connector using the `requests` library. It simulates sending multiple user messages to the chatbot API:

```python
import requests

class MillionBotConnector:
    def __init__(self):
        self.base_url = "https://api.1millionbot.com"
        self.path = "/api/public/messages"
        self.headers = {
            "Content-Type": "application/json",
            "Authorization": "64e5d1af081211d24e2cfecd"
        }
        self.static_payload = {
            "conversation": "6977595c744ccf01954b8797",
            "sender_type": "User",
            "sender": "6977595c3e4824018e496b0c",
            "bot": "64e5d1af081211d24e2cfec8",
            "language": "es",
            "url": "https://saic.malaga.eu/"
        }

    def send_message(self, user_msg):
        payload = self.static_payload.copy()
        payload["message"] = {"text": user_msg}
        response = requests.post(
            self.base_url + self.path,
            headers=self.headers,
            json=payload
        )
        if response.status_code == 200:
            data = response.json()
            # Extract the bot reply from response_path: response[0].text
            if "response" in data and len(data["response"]) > 0:
                return data["response"][0].get("text", "")
            else:
                return "No response text found."
        else:
            return f"Error: {response.status_code} - {response.text}"

if __name__ == "__main__":
    bot = MillionBotConnector()
    messages = [
        "hola",
        "¿Cómo estás?",
        "¿Qué puedes hacer?",
        "Gracias, adiós"
    ]

    for msg in messages:
        reply = bot.send_message(msg)
        print(f"User: {msg}")
        print(f"Bot: {reply}")
        print("-" * 40)
```

This script:
- Defines a connector class with the fixed parts of the request.
- Sends multiple user messages in sequence.
- Prints the bot's reply for each message.
