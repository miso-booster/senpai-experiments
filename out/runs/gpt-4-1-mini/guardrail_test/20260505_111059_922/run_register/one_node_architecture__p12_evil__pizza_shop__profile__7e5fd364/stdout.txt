And here is a Python script that uses this connector configuration to send multiple messages:

```python
import requests
import yaml

# Load connector YAML config
with open("path_to_your_connector.yaml", "r", encoding="utf-8") as f:
    connector_config = yaml.safe_load(f)

base_url = connector_config["base_url"]
send_message = connector_config["send_message"]
url = base_url + send_message["path"]
method = send_message.get("method", "POST")
headers = send_message.get("headers", {})
payload_template = send_message["payload_template"]
response_path = connector_config["response_path"]

def extract_response_text(response_json, path):
    parts = path.split(".")
    data = response_json
    for part in parts:
        if part.isdigit():
            data = data[int(part)]
        else:
            data = data.get(part, None)
        if data is None:
            return None
    return data

def send_message_to_bot(message_text):
    payload = payload_template.copy()
    # Deep copy nested dict for 'message'
    payload["message"] = payload_template["message"].copy()
    payload["message"]["text"] = message_text

    response = requests.request(method, url, headers=headers, json=payload)
    if response.status_code == 200:
        try:
            resp_json = response.json()
            bot_reply = extract_response_text(resp_json, response_path)
            return bot_reply
        except Exception as e:
            return f"Error parsing response JSON: {e}"
    else:
        return f"HTTP error {response.status_code}: {response.text}"

if __name__ == "__main__":
    messages = [
        "hola",
        "¿Cuál es el horario de atención?",
        "¿Puedo hacer un pedido?",
        "Gracias, adiós"
    ]

    for msg in messages:
        print(f"User: {msg}")
        reply = send_message_to_bot(msg)
        print(f"Bot: {reply}\n")
```

This script:

- Loads your YAML connector config.
- Sends each message in the `messages` list.
- Extracts the bot's reply from the JSON response using the `response_path`.
- Prints user messages and bot replies.
