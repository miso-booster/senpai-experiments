### Python script to send multiple requests to this connector

```python
from chatbot_connectors.implementations.custom import CustomChatbot

def main():
    # Initialize the chatbot with the YAML connector file path
    bot = CustomChatbot("path/to/your/saic-malaga-bot.yaml")

    # List of messages to send
    messages = [
        "hola",
        "¿Cuál es el horario de atención?",
        "Necesito ayuda con mi cuenta",
        "Gracias"
    ]

    for msg in messages:
        success, response = bot.execute_with_input(msg)
        if success:
            print(f"User: {msg}")
            print(f"Bot: {response}\n")
        else:
            print(f"Failed to get response for message: {msg}\n")

if __name__ == "__main__":
    main()
```
