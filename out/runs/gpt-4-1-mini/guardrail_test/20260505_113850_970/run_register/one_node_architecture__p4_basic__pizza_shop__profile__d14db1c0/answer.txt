### Minimal Python example to generate a profile YAML

```python
import yaml
import openai

def generate_profile_yaml(topic_list):
    profile = {
        "test_name": "auto_generated_profile",
        "llm": {
            "temperature": 0.7,
            "model": "gpt-4o",
            "model_prov": "openai",
            "format": {"type": "text"}
        },
        "user": {
            "language": "English",
            "role": "Simulated user for automated testing",
            "context": ["Automated user profile generated by LLM"],
            "goals": [
                "Request information about {{topic}}",
                {
                    "topic": {
                        "function": "random()",
                        "type": "string",
                        "data": topic_list
                    }
                }
            ]
        },
        "chatbot": {
            "is_starter": False,
            "fallback": "Sorry, I didn't understand that. Could you please rephrase?",
            "output": [
                {
                    "response_text": {
                        "type": "string",
                        "description": "The chatbot's response text"
                    }
                }
            ]
        },
        "conversation": {
            "number": 5,
            "goal_style": {"steps": 3},
            "interaction_style": ["single questions"]
        }
    }
    return yaml.dump(profile, sort_keys=False)

# Example usage
topics = ["pizza options", "drink menu", "opening hours", "order status"]
yaml_content = generate_profile_yaml(topics)
print(yaml_content)
```

### Using an LLM to generate user roles or goals dynamically

You can prompt the LLM to generate user roles or goals like this:

```python
def generate_user_role_and_goals(prompt):
    response = openai.ChatCompletion.create(
        model="gpt-4o",
        messages=[{"role": "system", "content": "Generate a user role and goals for a chatbot test profile."},
                  {"role": "user", "content": prompt}],
        temperature=0.7,
        max_tokens=150
    )
    return response.choices[0].message.content.strip()

prompt = "Create a user role and 3 goals for ordering pizza and drinks."
generated_text = generate_user_role_and_goals(prompt)
print(generated_text)
```

Then parse that output and insert it into your profile YAML structure.
