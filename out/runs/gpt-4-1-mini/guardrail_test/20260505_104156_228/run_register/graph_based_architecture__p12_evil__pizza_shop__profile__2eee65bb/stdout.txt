### Python Script to send multiple requests using this connector

```python
from chatbot_connectors.implementations.custom import CustomChatbot

def main():
    # Path to the YAML connector file you saved (adjust path as needed)
    connector_path = "path/to/saic-malaga-bot.yml"

    # Initialize the chatbot connector
    bot = CustomChatbot(connector_path)

    # List of messages to send
    messages = [
        "hola",
        "¿Cuál es el horario de atención?",
        "Necesito ayuda con mi cuenta",
        "Gracias, eso es todo"
    ]

    for msg in messages:
        success, response = bot.execute_with_input(msg)
        if success:
            print(f"User: {msg}")
            print(f"Bot: {response}\n")
        else:
            print(f"Failed to get response for message: {msg}")

if __name__ == "__main__":
    main()
```
