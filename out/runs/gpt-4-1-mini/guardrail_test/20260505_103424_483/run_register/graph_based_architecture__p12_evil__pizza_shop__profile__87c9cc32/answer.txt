### Now, here is a complete Python script to send multiple requests to this connector:

```python
from chatbot_connectors.implementations.custom import CustomChatbot

def main():
    # Load the connector YAML file (adjust the path if needed)
    connector_path = "path/to/your/saic-malaga-bot.yaml"
    bot = CustomChatbot(connector_path)

    # List of messages to send
    messages = [
        "hola",
        "¿Cuál es el horario de atención?",
        "¿Dónde está la oficina principal?",
        "Gracias"
    ]

    for msg in messages:
        success, response = bot.execute_with_input(msg)
        if success:
            print(f"User: {msg}")
            print(f"Bot: {response}\n")
        else:
            print(f"Failed to get response for message: {msg}")

if __name__ == "__main__":
    main()
```
