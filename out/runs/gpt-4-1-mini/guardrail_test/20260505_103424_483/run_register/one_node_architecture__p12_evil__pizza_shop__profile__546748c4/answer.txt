And here is a complete Python script that uses this connector YAML to send multiple messages:

```python
import yaml
import requests

class CustomChatbot:
    def __init__(self, config_path):
        with open(config_path, 'r', encoding='utf-8') as f:
            self.config = yaml.safe_load(f)
        self.base_url = self.config['base_url']
        self.send_message_conf = self.config['send_message']
        self.headers = self.send_message_conf.get('headers', {})
        self.path = self.send_message_conf['path']
        self.method = self.send_message_conf.get('method', 'POST').upper()
        self.payload_template = self.send_message_conf['payload_template']
        self.response_path = self.config['response_path']

    def _extract_response(self, json_response):
        parts = self.response_path.split('.')
        data = json_response
        for part in parts:
            if part.isdigit():
                data = data[int(part)]
            else:
                data = data.get(part)
            if data is None:
                break
        return data

    def send(self, user_msg):
        payload = self._build_payload(user_msg)
        url = self.base_url + self.path
        if self.method == 'POST':
            response = requests.post(url, json=payload, headers=self.headers)
        else:
            response = requests.get(url, params=payload, headers=self.headers)
        if response.status_code == 200:
            json_resp = response.json()
            return True, self._extract_response(json_resp)
        else:
            return False, f"HTTP {response.status_code}: {response.text}"

    def _build_payload(self, user_msg):
        # Recursively replace "{user_msg}" in the payload template
        def replace_text(obj):
            if isinstance(obj, dict):
                return {k: replace_text(v) for k, v in obj.items()}
            elif isinstance(obj, list):
                return [replace_text(i) for i in obj]
            elif isinstance(obj, str):
                return obj.replace("{user_msg}", user_msg)
            else:
                return obj
        return replace_text(self.payload_template)

if __name__ == "__main__":
    # Path to your connector YAML file
    connector_yaml_path = "path/to/your/connector.yaml"

    bot = CustomChatbot(connector_yaml_path)

    messages = [
        "hola",
        "¿Cuál es el horario de atención?",
        "¿Puedo hacer una reserva?",
        "Gracias, adiós"
    ]

    for msg in messages:
        success, response = bot.send(msg)
        if success:
            print(f"User: {msg}\nBot: {response}\n")
        else:
            print(f"Failed to get response for '{msg}': {response}\n")
```
