And here is a complete Python script that uses this connector to send multiple messages sequentially:

```python
import requests
import time

class OneMillionBotConnector:
    def __init__(self):
        self.base_url = "https://api.1millionbot.com"
        self.path = "/api/public/messages"
        self.headers = {
            "Content-Type": "application/json",
            "Authorization": "64e5d1af081211d24e2cfecd",
            "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64; rv:147.0) Gecko/20100101 Firefox/147.0",
            "Accept": "*/*",
            "Accept-Language": "en-US,en;q=0.9",
            "Accept-Encoding": "gzip, deflate, br, zstd",
            "Referer": "https://saic.malaga.eu/",
            "Origin": "https://saic.malaga.eu",
            "Sec-GPC": "1",
            "Connection": "keep-alive",
            "Sec-Fetch-Dest": "empty",
            "Sec-Fetch-Mode": "cors",
            "Sec-Fetch-Site": "cross-site",
            "Priority": "u=4",
            "Pragma": "no-cache",
            "Cache-Control": "no-cache"
        }
        self.payload_base = {
            "conversation": "6977595c744ccf01954b8797",
            "sender_type": "User",
            "sender": "6977595c3e4824018e496b0c",
            "bot": "64e5d1af081211d24e2cfec8",
            "language": "es",
            "url": "https://saic.malaga.eu/",
            "message": {
                "text": ""
            }
        }

    def send_message(self, user_msg):
        payload = self.payload_base.copy()
        payload["message"]["text"] = user_msg
        response = requests.post(
            self.base_url + self.path,
            headers=self.headers,
            json=payload
        )
        if response.ok:
            try:
                data = response.json()
                # Extract the bot's reply text from the response
                reply = data.get("response", [{}])[0].get("text", "")
                return True, reply
            except Exception as e:
                return False, f"Failed to parse response JSON: {e}"
        else:
            return False, f"HTTP error {response.status_code}: {response.text}"

if __name__ == "__main__":
    bot = OneMillionBotConnector()
    messages_to_send = [
        "hola",
        "¿Cuál es el horario de atención?",
        "¿Qué tipos de pizzas ofrecen?",
        "Quiero hacer un pedido para recoger.",
        "Gracias, eso es todo."
    ]

    for msg in messages_to_send:
        success, reply = bot.send_message(msg)
        if success:
            print(f"User: {msg}")
            print(f"Bot: {reply}\n")
        else:
            print(f"Error sending message '{msg}': {reply}\n")
        time.sleep(1)  # polite delay between messages
```
