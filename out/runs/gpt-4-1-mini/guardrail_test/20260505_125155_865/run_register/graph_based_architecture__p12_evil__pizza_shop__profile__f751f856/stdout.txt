### Python script to send multiple requests to this connector

```python
import yaml
import requests

class CustomChatbot:
    def __init__(self, config_path):
        with open(config_path, 'r', encoding='utf-8') as f:
            self.config = yaml.safe_load(f)
        self.base_url = self.config['base_url']
        self.path = self.config['send_message']['path']
        self.method = self.config['send_message'].get('method', 'POST').upper()
        self.headers = self.config['send_message'].get('headers', {})
        self.payload_template = self.config['send_message']['payload_template']
        self.response_path = self.config['response_path']

    def _extract_response(self, response_json):
        keys = self.response_path.split('.')
        data = response_json
        for key in keys:
            if key.isdigit():
                data = data[int(key)]
            else:
                data = data.get(key)
            if data is None:
                return None
        return data

    def send_message(self, user_msg):
        # Prepare payload by replacing {user_msg}
        def replace_msg(obj):
            if isinstance(obj, dict):
                return {k: replace_msg(v) for k, v in obj.items()}
            elif isinstance(obj, list):
                return [replace_msg(i) for i in obj]
            elif isinstance(obj, str):
                return obj.replace("{user_msg}", user_msg)
            else:
                return obj

        payload = replace_msg(self.payload_template)

        url = self.base_url + self.path
        if self.method == 'POST':
            response = requests.post(url, json=payload, headers=self.headers)
        else:
            response = requests.get(url, params=payload, headers=self.headers)

        if response.status_code == 200:
            try:
                response_json = response.json()
                return True, self._extract_response(response_json)
            except Exception as e:
                return False, f"JSON parse error: {e}"
        else:
            return False, f"HTTP error {response.status_code}: {response.text}"

if __name__ == "__main__":
    # Path to your YAML connector file
    connector_path = "saic-malaga-bot.yaml"  # Save the YAML content above in this file

    bot = CustomChatbot(connector_path)

    messages = [
        "hola",
        "¿Cuál es el horario de atención?",
        "¿Dónde está la oficina principal?",
        "Gracias por tu ayuda"
    ]

    for msg in messages:
        success, response = bot.send_message(msg)
        if success:
            print(f"User: {msg}")
            print(f"Bot: {response}\n")
        else:
            print(f"Error sending message '{msg}': {response}\n")
```
