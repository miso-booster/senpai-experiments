  goals:
    - "Preguntar cuál es el número de teléfono del CAU."
    - "Preguntar cuál es el correo electrónico de contacto del CAU."
    - "Preguntar cuál es la dirección física del CAU."
    - "Preguntar cuáles son los horarios de atención del CAU."

chatbot:
  is_starter: false
  fallback: "Por favor, ¿me lo puedes decir con otras palabras? Respondo a preguntas dirigidas al Centro de Atención a Usuarios del servicio de Tecnologías de la Información."

        description: "Número de teléfono del CAU"

        type: string
        description: "Correo electrónico del CAU"
    - physical_address:
        type: string

  interaction_style:
