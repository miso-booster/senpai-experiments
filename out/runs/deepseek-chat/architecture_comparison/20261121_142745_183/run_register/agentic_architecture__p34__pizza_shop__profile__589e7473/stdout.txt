```yaml
test_name: "pizza_order_custom_then_predefined"

llm:
  temperature: 0.8
  model: gpt-4o-mini
  model_prov: openai
  format:
    type: text

user:
  language: English
  role: >
    You are a customer ordering a pizza over the phone. You start by trying to
    build a custom pizza from scratch, but partway through the conversation you
    change your mind and decide to order a predefined pizza from the menu instead.
  context:
    - personality: personalities/conversational-user.yml
    - your name is Jon Doe
    - "You first want a custom pizza, but you get unsure about the toppings and
       eventually prefer a ready-made pizza from the menu."
  goals:
    - "a {{size}} custom pizza with {{toppings}}"
    - "Actually, I changed my mind — I'd rather have the {{predefined_pizza}} pizza instead"
    - "{{cans}} cans of {{drink}}"
    - how much will it cost
    - how long will it take to be ready

    - size:
        function: pairwise()
        type: string
        data:
          - small
          - medium
          - big
          - extra large

    - toppings:
        function: pairwise()
        type: string
        data:
          - cheese
          - mushrooms
          - pepperoni
          - olives

    - predefined_pizza:
        function: another()
        type: string
        data:
          - Margherita
          - Pepperoni
          - Four Cheese
          - Vegetarian
          - Hawaiian

    - cans:
        function: forward(drink)
        type: int
        data:
          min: 1
          max: 6
          step: 1

    - drink:
        function: forward()
        type: string
        data:
          - Sprite
          - Coca-Cola
          - Orange Fanta

chatbot:
  is_starter: false
  fallback: "I'm sorry, it's a little loud in my pizza shop, can you say that again?"
  output:
    - final_pizza:
        type: string
        description: "The pizza the customer ultimately ordered (the predefined one, not the custom one)"
    - price:
        type: currency
        description: "The final price of the pizza order"
    - time:
        type: time("hh:mm:ss")
        description: "The estimated time for the pizza to be ready"
    - order_id:
        type: string
        description: "The order identifier"

conversation:
  number: 2
  goal_style:
    steps: 6
    max_cost: 0.1
  interaction_style:
    - change your mind
    - random:
        - make spelling mistakes
        - long phrases
```
