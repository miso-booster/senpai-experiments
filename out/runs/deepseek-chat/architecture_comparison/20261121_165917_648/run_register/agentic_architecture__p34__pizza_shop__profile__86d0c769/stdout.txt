```yaml
test_name: "pizza_order_custom_then_predefined"

llm:
  temperature: 0.8
  model: gpt-4o-mini
  model_prov: openai
  format:
    type: text

user:
  language: English
  role: >
    You are a customer ordering a pizza over the phone. You start by asking
    for a custom pizza with your own chosen size and toppings, but partway
    through the conversation you change your mind and instead ask for a
    predefined pizza from the shop's menu.
  context:
    - personality: personalities/conversational-user.yml
    - your name is Jon Doe
    - "You first want a custom pizza, then you switch to a menu pizza."
  goals:
    - "I'd like a {{size}} custom pizza with {{toppings}}"
    - "Actually, forget the custom one — I'll take a {{pizza_type}} pizza instead"
    - "{{cans}} cans of {{drink}}"
    - how much will it cost
    - how long will it take to be ready

    - size:
        function: pairwise()
        type: string
        data:
          - small
          - medium
          - big
          - extra large

    - toppings:
        function: pairwise()
        type: string
        data:
          - cheese
          - mushrooms
          - pepperoni

    - pizza_type:
        function: another()
        type: string
        data:
          - Margherita
          - Pepperoni
          - Four Cheese
          - Vegetarian
          - any(3 classic pizza menu items)

    - cans:
        function: forward(drink)
        type: int
        data:
          min: 1
          max: 7
          step: 1

    - drink:
        function: forward()
        type: string
        data:
          - Sprite
          - Coke
          - Orange Fanta

chatbot:
  is_starter: false
  fallback: "I'm sorry it's a little loud in my pizza shop, can you say that again?"
  output:
    - price:
        type: currency
        description: The final price of the pizza order
    - time:
        type: time("hh:mm:ss")
        description: The estimated time for the pizza to be ready
    - order_id:
        type: string
        description: The order ID

conversation:
  number: 2
  goal_style:
    steps: 6
  interaction_style:
    - change your mind
    - random:
      - long phrases
      - make spelling mistakes
```

- **`goal_style.steps: 6`** gives enough room for the custom request, the switch, the drinks, and the price/ETA questions.
