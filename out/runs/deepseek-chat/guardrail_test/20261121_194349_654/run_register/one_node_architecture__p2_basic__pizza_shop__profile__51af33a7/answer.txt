test_name: "bike_shop_appointment_booking"

llm:

  model: gpt-4o-mini
  model_prov: openai

        type: string
        data:

        function: random()
        type: string
        data: ["10:00", "14:00", "16:30"]

chatbot:
  is_starter: false
  fallback: "Sorry, I didn't catch that. Could you rephrase?"
  output:

conversation:

  goal_style:
