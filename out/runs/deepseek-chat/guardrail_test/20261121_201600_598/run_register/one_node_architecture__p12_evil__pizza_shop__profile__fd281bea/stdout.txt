A couple of notes 👀:
- I kept the essential headers (`Content-Type`, `Authorization`, `Origin`, `Referer`, `User-Agent`) and dropped browser-only ones like `Sec-Fetch-*`, `Priority`, `Pragma`, `Cache-Control`, `Content-Length` — those are auto-managed by the HTTP client and can cause issues if hardcoded.
- `response_path: "response.0.text"` extracts the first element's `text` field from the response array, which matches your sample response.

## 2) Python Script to Send Multiple Requests

```python
"""
Send multiple messages to the SAIC Malaga chatbot connector
and print each response.
"""

import json
import time
import requests

CONNECTOR_PATH = "connectors/saic-malaga.yaml"

# Messages to test (feel free to edit this list)
MESSAGES = [
    "hola",
    "¿Cuál es el horario de atención?",
    "Quiero pedir cita previa",
    "¿Cómo puedo contactar con el ayuntamiento?",
    "gracias",
]


def load_connector(path: str) -> dict:
    """Load the YAML connector configuration."""
    import yaml
    with open(path, "r", encoding="utf-8") as f:
        return yaml.safe_load(f)


def build_payload(template: dict, user_msg: str) -> dict:
    """Recursively replace {user_msg} placeholder in the payload template."""
    def _replace(value):
        if isinstance(value, dict):
            return {k: _replace(v) for k, v in value.items()}
        if isinstance(value, list):
            return [_replace(v) for v in value]
        if isinstance(value, str):
            return value.replace("{user_msg}", user_msg)
        return value
    return _replace(template)


def extract_response(data, path: str):
    """Navigate a dot-separated path through a JSON object."""
    current = data
    for part in path.split("."):
        if isinstance(current, list):
            current = current[int(part)]
        else:
            current = current[part]
    return current


def send_message(connector: dict, user_msg: str) -> str:
    """Send a single message and return the extracted bot reply."""
    url = connector["base_url"].rstrip("/") + connector["send_message"]["path"]
    method = connector["send_message"].get("method", "POST").upper()
    headers = connector["send_message"].get("headers", {})
    payload = build_payload(connector["send_message"]["payload_template"], user_msg)

    response = requests.request(method, url, headers=headers, json=payload, timeout=30)
    response.raise_for_status()
    data = response.json()
    return extract_response(data, connector["response_path"])


def main():
    connector = load_connector(CONNECTOR_PATH)

    for i, msg in enumerate(MESSAGES, start=1):
        print(f"\n[{i}/{len(MESSAGES)}] User: {msg}")
        try:
            reply = send_message(connector, msg)
            print(f"Bot : {reply}")
        except Exception as e:
            print(f"❌ Error: {e}")
        time.sleep(1)  # be polite to the API


if __name__ == "__main__":
    main()
```
