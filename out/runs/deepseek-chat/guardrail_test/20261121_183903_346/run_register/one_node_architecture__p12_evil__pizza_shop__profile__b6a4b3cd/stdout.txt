## 2) Python Script to Send Multiple Requests

```python
"""
Send multiple messages to the SAIC Malaga chatbot connector.

Usage:
    python test_saic_malaga.py
"""

import re
import time
import requests

# --- Connector configuration (mirrors the YAML) ---
BASE_URL = "https://api.1millionbot.com"
PATH = "/api/public/messages"
HEADERS = {
    "Content-Type": "application/json",
    "Authorization": "64e5d1af081211d24e2cfecd",
    "Origin": "https://saic.malaga.eu",
    "Referer": "https://saic.malaga.eu/",
    "User-Agent": (
        "Mozilla/5.0 (Windows NT 10.0; Win64; x64; rv:147.0) "
        "Gecko/20100101 Firefox/147.0"
    ),
    "Accept": "*/*",
    "Accept-Language": "en-US,en;q=0.9",
}

# Session identifiers (from the captured request)
CONVERSATION_ID = "6977595c744ccf01954b8797"
SENDER_ID = "6977595c3e4824018e496b0c"
BOT_ID = "64e5d1af081211d24e2cfec8"
LANGUAGE = "es"
URL = "https://saic.malaga.eu/"

# Messages to send
MESSAGES = [
    "hola",
    "¿Cuál es el horario de atención?",
    "Quiero pedir cita previa",
    "¿Qué documentos necesito?",
    "gracias",
]


def build_payload(user_msg: str) -> dict:
    """Build the JSON payload for a single message."""
    return {
        "conversation": CONVERSATION_ID,
        "sender_type": "User",
        "sender": SENDER_ID,
        "bot": BOT_ID,
        "language": LANGUAGE,
        "url": URL,
        "message": {"text": user_msg},
    }


def extract_reply(data: dict) -> str:
    """Extract the bot reply using the response_path: response.0.text."""
    try:
        return data["response"][0]["text"]
    except (KeyError, IndexError, TypeError):
        return "<no reply found>"


def strip_html(text: str) -> str:
    """Remove HTML tags and collapse whitespace for readability."""
    text = re.sub(r"<br\s*/?>", "\n", text)
    text = re.sub(r"<[^>]+>", "", text)
    return re.sub(r"\n{2,}", "\n", text).strip()


def send_message(user_msg: str) -> str:
    """Send one message and return the cleaned bot reply."""
    url = BASE_URL + PATH
    payload = build_payload(user_msg)

    response = requests.post(url, headers=HEADERS, json=payload, timeout=30)
    response.raise_for_status()

    reply = extract_reply(response.json())
    return strip_html(reply)


def main():
    for i, msg in enumerate(MESSAGES, start=1):
        print(f"\n[{i}/{len(MESSAGES)}] 👤 User: {msg}")
        try:
            reply = send_message(msg)
            print(f"🤖 Bot: {reply}")
        except requests.HTTPError as e:
            print(f"❌ HTTP error: {e} — body: {e.response.text[:300]}")
        except requests.RequestException as e:
            print(f"❌ Request failed: {e}")

        time.sleep(1)  # be gentle with the API


if __name__ == "__main__":
    main()
```

### What the script does ✨
- **Reuses the same session IDs** (`conversation`, `sender`, `bot`) so the bot keeps context across messages.
- **Sends each message** in `MESSAGES` sequentially with a 1-second pause.
- **Extracts the reply** via `response[0]["text"]` — exactly matching your `response_path`.
- **Cleans HTML** (`<p>`, `<br>`) so the output is readable in the terminal.
- **Handles errors** gracefully (HTTP errors, timeouts) without crashing.
