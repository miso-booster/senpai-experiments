## 2) Python Script to Send Multiple Requests

```python
"""
Send multiple test messages to the SAIC Malaga chatbot connector.

Usage:
    python test_saic_malaga.py

Requires:
    pip install requests
"""

import json
import time
import requests


CONNECTOR_CONFIG = {
    "base_url": "https://api.1millionbot.com",
    "path": "/api/public/messages",
    "headers": {
        "Content-Type": "application/json",
        "Authorization": "64e5d1af081211d24e2cfecd",
        "Origin": "https://saic.malaga.eu",
        "Referer": "https://saic.malaga.eu/",
        "User-Agent": (
            "Mozilla/5.0 (Windows NT 10.0; Win64; x64; rv:147.0) "
            "Gecko/20100101 Firefox/147.0"
        ),
        "Accept": "*/*",
        "Accept-Language": "en-US,en;q=0.9",
    },
    "payload_template": {
        "conversation": "6977595c744ccf01954b8797",
        "sender_type": "User",
        "sender": "6977595c3e4824018e496b0c",
        "bot": "64e5d1af081211d24e2cfec8",
        "language": "es",
        "url": "https://saic.malaga.eu/",
        "message": {"text": "{user_msg}"},
    },
    "response_path": "response.0.text",
}


def build_payload(user_msg: str) -> dict:
    """Deep-copy the template and inject the user message."""
    payload = json.loads(json.dumps(CONNECTOR_CONFIG["payload_template"]))
    payload["message"]["text"] = user_msg
    return payload


def extract_response(data: dict, path: str):
    """Navigate a dot-separated path through a JSON object."""
    current = data
    for key in path.split("."):
        if isinstance(current, list):
            current = current[int(key)]
        else:
            current = current[key]
    return current


def send_message(user_msg: str) -> tuple[bool, str]:
    """Send one message to the chatbot and return (success, reply)."""
    url = CONNECTOR_CONFIG["base_url"] + CONNECTOR_CONFIG["path"]
    payload = build_payload(user_msg)

    try:
        resp = requests.post(
            url,
            headers=CONNECTOR_CONFIG["headers"],
            json=payload,
            timeout=30,
        )
        resp.raise_for_status()
        data = resp.json()
        reply = extract_response(data, CONNECTOR_CONFIG["response_path"])
        return True, reply
    except Exception as exc:
        return False, f"ERROR: {exc}"


def main():
    test_messages = [
        "hola",
        "¿Qué puedes hacer?",
        "Quiero pedir cita previa",
        "¿Cuál es el horario de atención?",
        "Necesito ayuda con un trámite",
    ]

    print(f"Sending {len(test_messages)} messages to SAIC Malaga chatbot\n")

    for i, msg in enumerate(test_messages, start=1):
        print(f"[{i}/{len(test_messages)}] USER: {msg}")
        success, reply = send_message(msg)

        if success:
            print(f"    BOT: {reply}\n")
        else:
            print(f"    {reply}\n")

        # Small pause to be polite to the API
        time.sleep(1)


if __name__ == "__main__":
    main()
```
