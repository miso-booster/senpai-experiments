## 2) Python Script to Send Multiple Requests

```python
"""
Send multiple messages to the SAIC Malaga connector (1millionbot API)
and print the bot's replies.

Usage:
    python send_multiple.py
"""

import json
import time
import requests

# --- Configuration -----------------------------------------------------------

BASE_URL = "https://api.1millionbot.com"
PATH = "/api/public/messages"

HEADERS = {
    "Content-Type": "application/json",
    "Authorization": "64e5d1af081211d24e2cfecd",
    "Origin": "https://saic.malaga.eu",
    "Referer": "https://saic.malaga.eu/",
    "User-Agent": (
        "Mozilla/5.0 (Windows NT 10.0; Win64; x64; rv:147.0) "
        "Gecko/20100101 Firefox/147.0"
    ),
    "Accept": "*/*",
    "Accept-Language": "en-US,en;q=0.9",
}

# Session identifiers (keep the same across messages to preserve context)
CONVERSATION_ID = "6977595c744ccf01954b8797"
SENDER_ID = "6977595c3e4824018e496b0c"
BOT_ID = "64e5d1af081211d24e2cfec8"
LANGUAGE = "es"
URL = "https://saic.malaga.eu/"

# Messages to send
MESSAGES = [
    "hola",
    "¿Cuál es el horario de atención?",
    "Quiero pedir cita previa",
    "gracias",
]

# --- Helpers -----------------------------------------------------------------

def build_payload(user_msg: str) -> dict:
    return {
        "conversation": CONVERSATION_ID,
        "sender_type": "User",
        "sender": SENDER_ID,
        "bot": BOT_ID,
        "language": LANGUAGE,
        "url": URL,
        "message": {"text": user_msg},
    }


def extract_reply(response_json: dict) -> str:
    """Navigate response_path = response.0.text safely."""
    try:
        return response_json["response"][0]["text"]
    except (KeyError, IndexError, TypeError):
        return "<no reply extracted>"


def send_message(user_msg: str) -> str:
    url = BASE_URL + PATH
    payload = build_payload(user_msg)

    try:
        r = requests.post(url, headers=HEADERS, json=payload, timeout=30)
        r.raise_for_status()
    except requests.RequestException as e:
        return f"<request error: {e}>"

    try:
        data = r.json()
    except json.JSONDecodeError:
        return f"<non-JSON response: {r.text[:200]}>"

    return extract_reply(data)


# --- Main --------------------------------------------------------------------

def main():
    for i, msg in enumerate(MESSAGES, start=1):
        print(f"\n[{i}/{len(MESSAGES)}] USER: {msg}")
        reply = send_message(msg)
        print(f"BOT: {reply}")
        time.sleep(1)  # be gentle with the API


if __name__ == "__main__":
    main()
```

### How it works 🚀
- **`build_payload`** mirrors the exact JSON body from your request, with `{user_msg}` swapped in.
- **`extract_reply`** follows the `response_path: response.0.text` logic safely.
- **`send_message`** handles HTTP errors and non-JSON responses gracefully.
- **`main`** loops through `MESSAGES`, keeping the same `conversation`/`sender` IDs so the bot retains context across turns.
