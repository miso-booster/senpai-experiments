Here's a quick taste of what a profile for that could look like:

```yaml
test_name: "bike_shop_appointment_scheduler"

llm:
  temperature: 0.7
  model: gpt-4o-mini
  model_prov: openai
  format:
    type: text

user:
  language: English
  role: >
    You are a customer trying to book a bike repair appointment at a local shop.
    Ask about availability, confirm a date and time, and request a confirmation.
  context:
    - "You prefer morning slots if possible."
    - "You have a road bike that needs a tune-up."
  goals:
    - "Book a {{service_type}} appointment on {{appointment_date}} at {{appointment_time}}"
    - "Ask for the appointment confirmation and total price"
    - service_type:
        function: forward()
        type: string
        data: [tune-up, brake adjustment, wheel truing, full service]
    - appointment_date:
        function: another()
        type: string
        data:
          - any(3 dates within the next 14 days)
    - appointment_time:
        function: pairwise()
        type: string
        data: ["09:00", "11:00", "14:00", "16:30"]

chatbot:
  is_starter: false
  fallback: "Sorry, I didn't catch that. Could you rephrase?"
  output:
    - appointment_date:
        type: date(day-month-year)
        description: "The confirmed appointment date"
    - appointment_time:
        type: time(hh:mm)
        description: "The confirmed appointment time"
    - total_price:
        type: currency
        description: "Total price for the service"
    - booking_id:
        type: string
        description: "Appointment confirmation / booking reference"

conversation:
  number: combinations(0.5, pairwise)
  goal_style:
    steps: 6
    max_cost: 0.1
  interaction_style:
    - random:
      - make spelling mistakes
      - single questions
      - long phrases
```
